    47→// ---- Search highlight helper ---------------------------------------------
    48→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    49→// match in a <mark> with a visible yellow background so the user can see why
    50→// the message was returned by the chat-search dialog.
    51→//
    52→// Supports multi-word queries: if the query contains spaces, each word is
    53→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    54→// and "Wasl" separately). Each word is regex-escaped so special characters
    55→// (`(`, `.`, `*`, …) are matched literally.
    56→//
    57→// Returns the original `text` (a single string node) when either the text or
    58→// query is empty, so plain messages with no active query render with zero
    59→// overhead and no behavioural change vs. before.
    60→function highlightText(text: string, query: string): React.ReactNode {
    61→  if (!text) return text
    62→  const q = query.trim()
    63→  if (!q) return text
    64→
    65→  // Split the query into individual words for multi-word highlighting.
    66→  // Filter out empty strings from multiple consecutive spaces.
    67→  const words = q.split(/\s+/).filter(Boolean)
    68→  if (words.length === 0) return text
    69→
    70→  // Build a single regex that matches ANY of the words (OR).
    71→  // Each word is regex-escaped so special characters are literal.
    72→  const escapedWords = words.map((w) =>
    73→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    74→  )
    75→  const pattern = escapedWords.join('|')
    76→  let regex: RegExp
    77→  try {
    78→    regex = new RegExp(`(${pattern})`, 'gi')
    79→  } catch {
    80→    // Malformed query — bail to plain text rather than crashing the bubble.
    81→    return text
    82→  }
    83→  // String.split with a capturing group keeps the matched delimiters in the
    84→  // result array, so odd indices are the query matches (highlight them) and
    85→  // even indices are the surrounding plain text.
    86→  const parts = text.split(regex)
    87→  if (parts.length === 1) {
    88→    // No matches — render as a single plain string node.
    89→    return text
    90→  }
    91→  return parts.map((part, i) =>
    92→    i % 2 === 1 ? (
    93→      <mark
    94→        key={i}
    95→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    96→      >
    97→        {part}
    98→      </mark>
    99→    ) : (
   100→      // Render empty strings as `null` so React doesn't complain about
   101→      // adjacent text separators. Non-empty even-indexed parts become plain
   102→      // string nodes (no wrapping element needed).
   103→      part || null
   104→    )
   105→  )
   106→}

   108→// ---- File-attachment content parser --------------------------------------
   109→// PDF / document / audio messages store their payload as a small JSON blob
   110→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   111→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   112→// resilient when the JSON shape isn't present.
   113→export function parseFileContent(
   114→  content: string
   115→): { url: string; name?: string; size?: number } | null {
   116→  if (!content) return null
   117→  try {
   118→    const parsed = JSON.parse(content)
   119→    if (
   120→      parsed &&
   121→      typeof parsed === 'object' &&
   122→      typeof parsed.url === 'string'
   123→    ) {
   124→      return {
   125→        url: parsed.url,
   126→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   127→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   128→      }
   129→    }
   130→  } catch {
   131→    // Not JSON — fall through to plain-URL detection.
   132→  }
   133→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   134→  if (content.startsWith('http') || content.startsWith('/')) {
   135→    return { url: content }
   136→  }
   137→  return null
   138→}

   140→export function formatFileSize(bytes: number): string {
   141→  if (bytes < 1024) return `${bytes} B`
   142→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   143→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   144→}

   146→// Derive a filename from a URL when the parsed payload has no `name`.
   147→function deriveFilenameFromUrl(url: string, fallback: string): string {
   148→  try {
   149→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   150→    const last = u.pathname.split('/').filter(Boolean).pop()
   151→    return last ? decodeURIComponent(last) : fallback
   152→  } catch {
   153→    return fallback
   154→  }

   157→// ---- Protection helpers -------------------------------------------------
   158→// Resolves whether a message is effectively protected from screenshot /
   159→// forwarding. The effective flag is:
   160→//   - the explicit `message.protected` value if it is a boolean
   161→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   162→//     the sender's settings here, so we just use the explicit value — the
   163→//     backend always resolves and persists it on send, so the frontend
   164→//     always sees the resolved boolean).
   165→//
   166→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   167→// is true only when the current user is the recipient of a protected
   168→// message and has NOT opted into "privacy always allow".
   169→function useProtectionState(message: ChatMessage) {
   170→  const me = useWaslStore((s) => s.user)
   171→  const isOwner = !!me && message.senderId === me.id
   172→  const alwaysAllow = !!me?.privacyAlwaysAllow
   173→  const isProtected = message.protected === true
   174→  const blocked = isProtected && !isOwner && !alwaysAllow
   175→  return { isProtected, isOwner, alwaysAllow, blocked }
   176→}

   178→// Record a screenshot/copy/save attempt to the audit log via the API and
   179→// show a toast. Idempotent — debounced per-message per-kind to avoid
   180→// spamming the server when the user holds down PrintScreen.
   181→const recentAttemptCache = new Map<string, number>()
   182→function recordAttempt(messageId: string, kind: string) {
   183→  const key = `${messageId}:${kind}`
   184→  const now = Date.now()
   185→  if (recentAttemptCache.has(key)) {
   186→    const last = recentAttemptCache.get(key)!
   187→    // Debounce: at most one record per 5 seconds per kind per message.
   188→    if (now - last < 5000) return
   189→  }
   190→  recentAttemptCache.set(key, now)
   191→  // Fire and forget
   192→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   193→    method: 'POST',
   194→    headers: { 'Content-Type': 'application/json' },
   195→    body: JSON.stringify({ kind }),
   196→  }).catch(() => {})
   197→}

   911→// Voice message playback bubble with a simple play/pause + waveform.
   912→function VoiceMessagePlayer({
   913→  src,
   914→  mine,
   915→  status,
   916→  createdAt,
   917→}: {
   918→  src: string
   919→  mine: boolean
   920→  status: string
   921→  createdAt: string
   922→}) {
   923→  const [playing, setPlaying] = useState(false)
   924→  const [progress, setProgress] = useState(0)
   925→  const [duration, setDuration] = useState(0)
   926→  const audioRef = useRef<HTMLAudioElement | null>(null)
   927→
   928→  function toggle() {
   929→    if (!audioRef.current) {
   930→      audioRef.current = new Audio(src)
   931→      audioRef.current.addEventListener('loadedmetadata', () => {
   932→        setDuration(audioRef.current?.duration || 0)
   933→      })
   934→      audioRef.current.addEventListener('timeupdate', () => {
   935→        setProgress(audioRef.current?.currentTime || 0)
   936→      })
   937→      audioRef.current.addEventListener('ended', () => {
   938→        setPlaying(false)
   939→        setProgress(0)
   940→      })
   941→    }
   942→    if (playing) {
   943→      audioRef.current.pause()
   944→      setPlaying(false)
   945→    } else {
   946→      audioRef.current.play()
   947→      setPlaying(true)
   948→    }
   949→  }
   950→
   951→  function formatTime(sec: number) {
   952→    if (!sec || Number.isNaN(sec)) return '0:00'
   953→    const m = Math.floor(sec / 60)
   954→    const s = Math.floor(sec % 60)
   955→    return `${m}:${String(s).padStart(2, '0')}`
   956→  }
   957→
   958→  const pct = duration > 0 ? (progress / duration) * 100 : 0
   959→  // Deterministic pseudo-waveform bars
   960→  const bars = Array.from({ length: 28 }, (_, i) => {
   961→    const h = 30 + Math.sin(i * 1.3 + src.length) * 20 + Math.cos(i * 2.1) * 15
   962→    return Math.max(10, Math.min(100, Math.abs(h)))
   963→  })
   964→
   965→  return (
   966→    <div className="flex items-center gap-2 min-w-[200px]">
   967→      <button
   968→        type="button"
   969→        onClick={toggle}
   970→        className="w-9 h-9 rounded-full bg-[var(--wasl-green)] text-white flex items-center justify-center shrink-0 hover:opacity-90"
   971→      >
   972→        {playing ? (
   973→          <Pause className="w-4 h-4" />
   974→        ) : (
   975→          <Play className="w-4 h-4 ml-0.5" />
   976→        )}
   977→      </button>
   978→      <div className="flex-1 min-w-0">
   979→        <div className="flex items-center gap-0.5 h-8">
   980→          {bars.map((h, i) => {
   981→            const barProgress = (i / bars.length) * 100
   982→            const active = barProgress < pct
   983→            return (
   984→              <div
   985→                key={i}
   986→                className={cn(
   987→                  'w-0.5 rounded-full transition-colors',
   988→                  active
   989→                    ? 'bg-[var(--wasl-green)]'
   990→                    : mine
   991→                    ? 'bg-foreground/30'
   992→                    : 'bg-foreground/20'
   993→                )}
   994→                style={{ height: `${h}%` }}
   995→              />
   996→            )
   997→          })}
   998→        </div>
   999→        <div className="flex items-center justify-between mt-0.5">
  1000→          <span className="text-[10px] text-foreground/60">
  1001→            {formatTime(playing ? progress : duration)}
  1002→          </span>
  1003→          <span className="inline-flex items-center gap-1 text-[10px] text-foreground/50">
  1004→            {formatChatTimestamp(createdAt)}
  1005→            {mine && <StatusTicks status={status} />}
  1006→          </span>
  1007→        </div>
  1008→      </div>
  1009→    </div>
  1010→  )
  1011→}

  1106→// ---- Audio URL card ------------------------------------------------------
  1107→// Renders an `<audio controls>` element for audio messages whose `content`
  1108→// is a URL (i.e. an uploaded .mp3/.wav/.m4a/.ogg). Voice notes that were
  1109→// recorded in-browser are stored as `data:audio/...` data URIs and are
  1110→// handled by the existing VoiceMessagePlayer above.
  1111→function AudioUrlCardContent({
  1112→  message,
  1113→  mine,
  1114→  blocked,
  1115→  isProtected,
  1116→  onOpenReadReceipts,
  1117→}: {
  1118→  message: ChatMessage
  1119→  mine: boolean
  1120→  blocked: boolean
  1121→  isProtected: boolean
  1122→  onOpenReadReceipts: () => void
  1123→}) {
  1124→  const parsed = parseFileContent(message.content)
  1125→  const url = parsed?.url
  1126→  const name =
  1127→    parsed?.name ||
  1128→    (url ? deriveFilenameFromUrl(url, 'Audio message') : 'Audio message')
  1129→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : undefined
  1130→
  1131→  if (!url) {
  1132→    // Malformed payload — fall back to a tiny "audio unavailable" notice
  1133→    // rather than crashing the whole bubble.
  1134→    return (
  1135→      <div className="rounded-lg p-2 max-w-[280px] text-xs text-muted-foreground">
  1136→        Audio unavailable
  1137→      </div>
  1138→    )
  1139→  }
  1140→
  1141→  return (
  1142→    <div className="rounded-lg overflow-hidden max-w-[300px] p-1">
  1143→      <div className="flex items-center gap-2 mb-1 min-w-0">
  1144→        <Music className="w-4 h-4 text-[var(--wasl-green)] shrink-0" />
  1145→        <span className="text-xs font-medium truncate flex-1 text-foreground" title={name}>
  1146→          {name}
  1147→        </span>
  1148→        {sizeLabel && (
  1149→          <span className="text-[10px] text-foreground/60 shrink-0">{sizeLabel}</span>
  1150→        )}
  1151→      </div>
  1152→      {/* `controlsList="nodownload noplaybackrate"` keeps the native UI tidy
  1153→          and discourages trivial screenshotting on protected bubbles (though
  1154→          the only true protection is the audit log). */}
  1155→      <audio
  1156→        controls
  1157→        src={url}
  1158→        className="w-full"
  1159→        controlsList="nodownload noplaybackrate"
  1160→        // Protected recipients can't interact with the audio player beyond
  1161→        // viewing the file card (mirrors the image's pointer-events:none
  1162→        // treatment above).
  1163→        style={blocked ? { pointerEvents: 'none', opacity: 0.6 } : undefined}
  1164→      />
  1165→      <div className="text-[10px] text-right text-foreground/60 mt-0.5 flex items-center justify-end gap-1">
  1166→        {isProtected && <Lock className="w-3 h-3 inline opacity-60" />}
  1167→        {formatChatTimestamp(message.createdAt)}
  1168→        {mine && <StatusTicks status={message.status} className="ml-1" onClick={onOpenReadReceipts} />}
  1169→      </div>
  1170→    </div>
  1171→  )

  1203→function StatusTicks({
  1204→  status,
  1205→  className,
  1206→  onClick,
  1207→}: {
  1208→  status: string
  1209→  className?: string
  1210→  onClick?: () => void
  1211→}) {
  1212→  if (status === 'sent') {
  1213→    return <Check className={cn('w-3.5 h-3.5 inline', className)} />
  1214→  }
  1215→  if (status === 'delivered') {
  1216→    return <CheckCheck className={cn('w-3.5 h-3.5 inline', className)} />
  1217→  }
  1218→  if (status === 'read') {
  1219→    // When onClick is provided, the read-ticks become a clickable button that
  1220→    // opens the read-receipts dialog (sender-only feature).
  1221→    if (onClick) {

  1223→        <button
  1224→          type="button"
  1225→          onClick={(e) => {
  1226→            e.stopPropagation()

  1229→          className="inline-flex items-center hover:scale-110 transition-transform"
  1230→          title="Read — click to see who read this message"
  1231→          aria-label="Read — click to see details"

  1233→          <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />
  1234→        </button>
  1235→      )
  1236→    }
  1237→    return (
  1238→      <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />
  1239→    )

  1241→  return <Clock className={cn('w-3.5 h-3.5 inline', className)} />

  1244→function groupReactions(reactions: Reaction[]) {
  1245→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1246→  for (const r of reactions) {
  1247→    const ex = map.get(r.emoji)
  1248→    if (ex) {
  1249→      ex.count++
  1250→      ex.users.push(r.userId)
  1251→    } else {
  1252→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1253→    }
  1254→  }
  1255→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
  1256→}
