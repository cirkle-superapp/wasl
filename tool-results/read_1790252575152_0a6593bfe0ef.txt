   301→    function onStatus(payload: {
   302→      conversationId: string

   310→      // update the per-recipient read summary (readCount / readByEveryone) on
   311→      // the affected outgoing messages. The server doesn't push the full
   312→      // per-recipient breakdown over the socket — we derive it locally by
   313→      // treating each 'read' event from a distinct participant as +1 read.
   314→      // `bumpMessageReadCount` is a no-op for messages without `totalRecipients`
   315→      // (incoming messages / older state), and clamps to the total so a repeat
   316→      // 'read' event from the same participant doesn't double-count.
   317→      if (payload.status === 'read' && payload.byUserId !== user?.id) {
   318→        bumpMessageReadCount(activeConversationId, payload.messageIds)
   319→      }
   320→    }

   426→    function onJumpToMessage(e: Event) {

   463→    function onMessageHiddenForMe(e: Event) {

   488→  // ---- Load older messages ---------------------------------------------------
   489→  async function loadMore() {
   490→    if (!activeConversationId || !oldestLoaded) return
   491→    setLoadingMore(true)
   492→    lastMessageHeightRef.current = scrollRef.current?.scrollHeight || 0
   493→    try {
   494→      const res = await fetch(
   495→        `/api/conversations/${activeConversationId}/messages?before=${encodeURIComponent(
   496→          oldestLoaded
   497→        )}&limit=${PAGE_SIZE}`,
   498→        { cache: 'no-store' }
   499→      )
   500→      if (!res.ok) return
   501→      const data = await res.json()
   502→      const msgs: ChatMessage[] = data.messages
   503→      if (msgs.length === 0) {
   504→        setHasMore(false)
   505→      } else {
   506→        prependMessages(activeConversationId, msgs)
   507→        msgs.forEach((m) => messageIdsRef.current.add(m.id))
   508→        setOldestLoaded(msgs[0].createdAt)
   509→        if (msgs.length < PAGE_SIZE) setHasMore(false)
   510→        // Preserve scroll position
   511→        requestAnimationFrame(() => {
   512→          if (scrollRef.current) {
   513→            const newHeight = scrollRef.current.scrollHeight
   514→            scrollRef.current.scrollTop =
   515→              newHeight - lastMessageHeightRef.current
   516→          }
   517→        })
   518→      }
   519→    } catch (e) {
   520→      console.error(e)
   521→    } finally {
   522→      setLoadingMore(false)
   523→    }
   524→  }

   694→      setBookmarked(messageId, !isCurrentlyBookmarked)

   696→        if (isCurrentlyBookmarked) {

   698→          // ids in the store (only message ids), so re-fetch the list and
   699→          // find the row matching this message.
   700→          const res = await fetch('/api/bookmarks', { cache: 'no-store' })

   716→          const res = await fetch('/api/bookmarks', {
   717→            method: 'POST',
   718→            headers: { 'Content-Type': 'application/json' },
   719→            body: JSON.stringify({ messageId }),

   722→            // Already bookmarked server-side — make sure store reflects that.
   723→            setBookmarked(messageId, true)
