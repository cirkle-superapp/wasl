   448→// ==========================================================================
   449→// Official Service Providers — banks, government ministries, utilities.
   450→// These are manually authenticated and approved by Wasl admin staff.
   451→// Once approved, they can broadcast official announcements to all users
   452→// within a specific country.
   453→//
   454→// Registration flow:
   455→//   1. User applies → uploads official registration documents + passes
   456→//      liveness test (selfie video) + ID document
   457→//   2. Wasl admin manually reviews → approves or rejects
   458→//   3. Once approved, the provider gets a verified badge + broadcast rights
   459→//
   460→// Broadcast flow:
   461→//   1. Provider creates an announcement (text/image)
   462→//   2. System delivers to ALL users whose phone country code matches
   463→//   3. Users see official announcements in a dedicated "Official" tab
   464→// ==========================================================================
   465→
   466→model ServiceProvider {
   467→  id                String   @id @default(cuid())
   468→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   469→  ownerId           String   // the user who registered the provider
   470→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   471→  description       String   @default("")
   472→  avatarPath        String?
   473→  avatarColor       String?
   474→  // Official registration details
   475→  officialName      String?  // legal entity name
   476→  registrationNumber String? // government registration number
   477→  taxNumber         String?
   478→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   479→  // All users whose phone starts with this code receive broadcasts
   480→  countryCode       String   @default("+20")
   481→  // Verification documents
   482→  registrationDocPath String?  // official registration document
   483→  idDocPath           String?  // ID of the authorized representative
   484→  livenessVideoPath  String?  // selfie video for liveness check
   485→  livenessVerified   Boolean  @default(false)
   486→  // Approval state — manually reviewed by Wasl admin
   487→  status            String   @default("pending") // pending | approved | rejected
   488→  rejectionReason   String?
   489→  verified          Boolean  @default(false)
   490→  verifiedAt        DateTime?
   491→  verifiedBy        String?  // admin user ID who approved
   492→  // Broadcast settings
   493→  canBroadcast      Boolean  @default(false)  // can send announcements
   494→  // Contact info
   495→  contactPhone      String?
   496→  contactEmail      String?
   497→  website           String?
   498→  createdAt         DateTime @default(now())
   499→  updatedAt         DateTime @updatedAt
   500→
   501→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   502→  broadcasts ServiceProviderMessage[]
   503→
   504→  @@index([ownerId])
   505→  @@index([countryCode])
   506→  @@index([status])
   507→  @@index([type])
   508→}
