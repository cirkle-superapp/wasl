   506→// ==========================================================================
   507→// Official Service Providers — banks, government ministries, utilities.
   508→// These are manually authenticated and approved by Wasl admin staff.
   509→// Once approved, they can broadcast official announcements to all users
   510→// within a specific country.
   511→//
   512→// Registration flow:
   513→//   1. User applies → uploads official registration documents + passes
   514→//      liveness test (selfie video) + ID document
   515→//   2. Wasl admin manually reviews → approves or rejects
   516→//   3. Once approved, the provider gets a verified badge + broadcast rights
   517→//
   518→// Broadcast flow:
   519→//   1. Provider creates an announcement (text/image)
   520→//   2. System delivers to ALL users whose phone country code matches
   521→//   3. Users see official announcements in a dedicated "Official" tab
   522→// ==========================================================================
   523→
   524→model ServiceProvider {
   525→  id                String   @id @default(cuid())
   526→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   527→  ownerId           String   // the user who registered the provider
   528→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   529→  description       String   @default("")
   530→  avatarPath        String?
   531→  avatarColor       String?
   532→  // Official registration details
   533→  officialName      String?  // legal entity name
   534→  registrationNumber String? // government registration number
   535→  taxNumber         String?
   536→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   537→  // All users whose phone starts with this code receive broadcasts
   538→  countryCode       String   @default("+20")
   539→  // Verification documents
   540→  registrationDocPath String?  // official registration document
   541→  idDocPath           String?  // ID of the authorized representative
   542→  livenessVideoPath  String?  // selfie video for liveness check
   543→  livenessVerified   Boolean  @default(false)
   544→  // Approval state — manually reviewed by Wasl admin
   545→  status            String   @default("pending") // pending | approved | rejected
   546→  rejectionReason   String?
   547→  verified          Boolean  @default(false)
   548→  verifiedAt        DateTime?
   549→  verifiedBy        String?  // admin user ID who approved
   550→  // Broadcast settings
   551→  canBroadcast      Boolean  @default(false)  // can send announcements
   552→  // Contact info
   553→  contactPhone      String?
   554→  contactEmail      String?
   555→  website           String?
   556→  createdAt         DateTime @default(now())
   557→  updatedAt         DateTime @updatedAt
   558→
   559→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   560→  broadcasts ServiceProviderMessage[]
   561→
   562→  @@index([ownerId])
   563→  @@index([countryCode])
   564→  @@index([status])
   565→  @@index([type])
   566→}

   617→/// Scheduled message — send at a future time (Cirkle-inspired scheduled-messages)
   618→/// Supports recurring schedules: `repeat` can be 'none' (one-time), 'daily',
   619→/// 'weekly', or 'monthly'. When a recurring message is sent, `scheduledFor`
   620→/// is advanced to the next occurrence and `sent`/`sentAt` are reset so the
   621→/// background scheduler picks it up again at the next time.
   622→model ScheduledMessage {
   623→  id             String   @id @default(cuid())
   624→  conversationId String
   625→  senderId       String
   626→  content        String
   627→  type           String   @default("text")
   628→  scheduledFor   DateTime
   629→  sent           Boolean  @default(false)
   630→  sentAt         DateTime?
   631→  // Recurring schedule: 'none' = one-time, 'daily', 'weekly', 'monthly'.
   632→  // When non-'none', the scheduler advances `scheduledFor` to the next
   633→  // occurrence after sending instead of marking `sent=true`.
   634→  repeat         String   @default("none")
   635→  // Optional end date for recurring messages. After this date, the schedule
   636→  // stops (message is marked sent=true so the scheduler skips it).
   637→  repeatUntil    DateTime?
   638→  createdAt      DateTime @default(now())
   639→
   640→  @@index([scheduledFor])
   641→  @@index([conversationId])
   642→  @@index([senderId])
   643→}
