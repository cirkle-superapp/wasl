   174→/// Cirkle-inspired Commit (AI-verified agreement) anchored to a conversation.
   175→/// Simplified subset for Wasl: type, parties (creator + counterparty),

   424→// ==========================================================================
   425→// Official Service Providers — banks, government ministries, utilities.
   426→// These are manually authenticated and approved by Wasl admin staff.
   427→// Once approved, they can broadcast official announcements to all users
   428→// within a specific country.
   429→//
   430→// Registration flow:
   431→//   1. User applies → uploads official registration documents + passes
   432→//      liveness test (selfie video) + ID document
   433→//   2. Wasl admin manually reviews → approves or rejects
   434→//   3. Once approved, the provider gets a verified badge + broadcast rights
   435→//
   436→// Broadcast flow:
   437→//   1. Provider creates an announcement (text/image)
   438→//   2. System delivers to ALL users whose phone country code matches
   439→//   3. Users see official announcements in a dedicated "Official" tab
   440→// ==========================================================================
   441→
   442→model ServiceProvider {
   443→  id                String   @id @default(cuid())
   444→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   445→  ownerId           String   // the user who registered the provider
   446→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   447→  description       String   @default("")
   448→  avatarPath        String?
   449→  avatarColor       String?
   450→  // Official registration details
   451→  officialName      String?  // legal entity name
   452→  registrationNumber String? // government registration number
   453→  taxNumber         String?
   454→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   455→  // All users whose phone starts with this code receive broadcasts
   456→  countryCode       String   @default("+20")
   457→  // Verification documents
   458→  registrationDocPath String?  // official registration document
   459→  idDocPath           String?  // ID of the authorized representative
   460→  livenessVideoPath  String?  // selfie video for liveness check
   461→  livenessVerified   Boolean  @default(false)
   462→  // Approval state — manually reviewed by Wasl admin
   463→  status            String   @default("pending") // pending | approved | rejected
   464→  rejectionReason   String?
   465→  verified          Boolean  @default(false)
   466→  verifiedAt        DateTime?
   467→  verifiedBy        String?  // admin user ID who approved
   468→  // Broadcast settings
   469→  canBroadcast      Boolean  @default(false)  // can send announcements
   470→  // Contact info
   471→  contactPhone      String?
   472→  contactEmail      String?
   473→  website           String?
   474→  createdAt         DateTime @default(now())
   475→  updatedAt         DateTime @updatedAt
   476→
   477→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   478→  broadcasts ServiceProviderMessage[]
   479→
   480→  @@index([ownerId])
   481→  @@index([countryCode])
   482→  @@index([status])
   483→  @@index([type])
   484→}

   693→/// Screenshot / capture attempt log — when a user attempts to screenshot or

   696→/// feature. `kind` is one of: `printscreen`, `copy`, `save`, `contextmenu`,
   697→/// `drag`, `forward`.
   698→model ScreenshotAttempt {
   699→  id        String   @id @default(cuid())
   700→  messageId String
   701→  reporterId String? // the user who reported the attempt (the recipient, often null for server-detected)
   702→  kind      String   @default("printscreen") // printscreen | copy | save | contextmenu | drag | forward
   703→  note      String?
   704→  createdAt DateTime @default(now())
   705→
   706→  message Message @relation(fields: [messageId], references: [id], onDelete: Cascade)
   707→  reporter User?  @relation("ScreenshotReporter", fields: [reporterId], references: [id], onDelete: SetNull)
   708→
   709→  @@index([messageId, createdAt])
   710→  @@index([reporterId])
   711→}
