   482→// ==========================================================================
   483→// Official Service Providers — banks, government ministries, utilities.
   484→// These are manually authenticated and approved by Wasl admin staff.
   485→// Once approved, they can broadcast official announcements to all users
   486→// within a specific country.
   487→//
   488→// Registration flow:
   489→//   1. User applies → uploads official registration documents + passes
   490→//      liveness test (selfie video) + ID document
   491→//   2. Wasl admin manually reviews → approves or rejects
   492→//   3. Once approved, the provider gets a verified badge + broadcast rights
   493→//
   494→// Broadcast flow:
   495→//   1. Provider creates an announcement (text/image)
   496→//   2. System delivers to ALL users whose phone country code matches
   497→//   3. Users see official announcements in a dedicated "Official" tab
   498→// ==========================================================================
   499→
   500→model ServiceProvider {
   501→  id                String   @id @default(cuid())
   502→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   503→  ownerId           String   // the user who registered the provider
   504→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   505→  description       String   @default("")
   506→  avatarPath        String?
   507→  avatarColor       String?
   508→  // Official registration details
   509→  officialName      String?  // legal entity name
   510→  registrationNumber String? // government registration number
   511→  taxNumber         String?
   512→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   513→  // All users whose phone starts with this code receive broadcasts
   514→  countryCode       String   @default("+20")
   515→  // Verification documents
   516→  registrationDocPath String?  // official registration document
   517→  idDocPath           String?  // ID of the authorized representative
   518→  livenessVideoPath  String?  // selfie video for liveness check
   519→  livenessVerified   Boolean  @default(false)
   520→  // Approval state — manually reviewed by Wasl admin
   521→  status            String   @default("pending") // pending | approved | rejected
   522→  rejectionReason   String?
   523→  verified          Boolean  @default(false)
   524→  verifiedAt        DateTime?
   525→  verifiedBy        String?  // admin user ID who approved
   526→  // Broadcast settings
   527→  canBroadcast      Boolean  @default(false)  // can send announcements
   528→  // Contact info
   529→  contactPhone      String?
   530→  contactEmail      String?
   531→  website           String?
   532→  createdAt         DateTime @default(now())
   533→  updatedAt         DateTime @updatedAt
   534→
   535→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   536→  broadcasts ServiceProviderMessage[]
   537→
   538→  @@index([ownerId])
   539→  @@index([countryCode])
   540→  @@index([status])
   541→  @@index([type])
   542→}

   593→/// Scheduled message — send at a future time (Cirkle-inspired scheduled-messages)
   594→/// Supports recurring schedules: `repeat` can be 'none' (one-time), 'daily',
   595→/// 'weekly', or 'monthly'. When a recurring message is sent, `scheduledFor`
   596→/// is advanced to the next occurrence and `sent`/`sentAt` are reset so the
   597→/// background scheduler picks it up again at the next time.
   598→model ScheduledMessage {
   599→  id             String   @id @default(cuid())
   600→  conversationId String
   601→  senderId       String
   602→  content        String
   603→  type           String   @default("text")
   604→  scheduledFor   DateTime
   605→  sent           Boolean  @default(false)
   606→  sentAt         DateTime?
   607→  // Recurring schedule: 'none' = one-time, 'daily', 'weekly', 'monthly'.
   608→  // When non-'none', the scheduler advances `scheduledFor` to the next
   609→  // occurrence after sending instead of marking `sent=true`.
   610→  repeat         String   @default("none")
   611→  // Optional end date for recurring messages. After this date, the schedule
   612→  // stops (message is marked sent=true so the scheduler skips it).
   613→  repeatUntil    DateTime?
   614→  createdAt      DateTime @default(now())
   615→
   616→  @@index([scheduledFor])
   617→  @@index([conversationId])
   618→  @@index([senderId])
   619→}
