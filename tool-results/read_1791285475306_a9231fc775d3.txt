   275→/// Cirkle-inspired Commit (AI-verified agreement) anchored to a conversation.
   276→/// Simplified subset for Wasl: type, parties (creator + counterparty),

   278→/// two-party signature state machine (pending -> active -> completed|disputed).
   279→model Commit {
   280→  id              String   @id @default(cuid())
   281→  conversationId  String
   282→  creatorId       String
   283→  counterpartyId  String
   284→  type            String   @default("price") // price|work|service|rental|group_buy
   285→  title           String
   286→  description     String   @default("")
   287→  amount          Float   @default(0)
   288→  currency        String   @default("SAR")
   289→  deadline        String?
   290→  conditions      String   @default("[]") // JSON-encoded string[]
   291→  status          String   @default("pending") // pending|active|completed|disputed
   292→  fairnessScore   Int      @default(0) // 0-100
   293→  fairnessNote    String   @default("")
   294→  hash            String
   295→  creatorSigned   Boolean  @default(true)
   296→  counterpartySigned Boolean @default(false)
   297→  creatorSignedAt  DateTime?
   298→  counterpartySignedAt DateTime?
   299→  completedAt     DateTime?
   300→  createdAt       DateTime @default(now())
   301→  updatedAt       DateTime @updatedAt
   302→
   303→  conversation Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
   304→  creator      User         @relation("CommitCreator", fields: [creatorId], references: [id], onDelete: Cascade)
   305→  counterparty User         @relation("CommitCounterparty", fields: [counterpartyId], references: [id], onDelete: Cascade)

   483→// Business accounts — Cirkle-inspired business feature.
   484→// A verified person can register a business. The business must be verified
   485→// (upload company registration + tax number + ID of someone on the
   486→// registration). Once verified, the owner can create public/private groups
   487→// and invite members. A business can have a hidden phone number that only
   488→// surfaces via the business search.
   489→// ==========================================================================
   490→
   491→model Business {
   492→  id              String   @id @default(cuid())
   493→  name            String
   494→  ownerId         String
   495→  // Public search handle (unique). Users find the business by this name.
   496→  description     String   @default("")
   497→  avatarPath      String?
   498→  avatarColor     String?
   499→  category        String?  // e.g. "Technology", "Food", "Customer Service"
   500→  // Verification
   501→  registrationDocPath String?  // company registration document
   502→  taxDocPath          String?  // tax number document
   503→  idDocPath           String?  // ID of someone on the registration
   504→  status          String   @default("pending") // pending | approved | rejected
   505→  rejectionReason String?
   506→  verifiedAt      DateTime?
   507→  // A hidden phone number that only surfaces via business search.
   508→  hiddenPhone     String?
   509→  hidePhone       Boolean  @default(false)
   510→  verified        Boolean  @default(false)
   511→  // ---- Privacy / message protection (business accounts) ---------------------
   512→  // Same semantics as User.defaultProtectMessages: when true, every message
   513→  // sent "as this business" is protected from screenshot/forwarding unless the
   514→  // individual message overrides it with `protected: false`.
   515→  defaultProtectMessages Boolean @default(false)
   516→  createdAt       DateTime @default(now())
   517→  updatedAt       DateTime @updatedAt
   518→
   519→  owner   User              @relation("BusinessOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   520→  members BusinessMember[]
   521→  groups  BusinessGroup[]
   522→
   523→  @@index([ownerId])
   524→}

  1018→// ==========================================================================
  1019→// Block — user-to-user blocking
  1020→//
  1021→// A blocks B → B can't send messages to A, can't see A's online/last-seen,
  1022→// can't add A to new groups. A can unblock at any time.
  1023→//
  1024→// Symmetric privacy: when A blocks B, B's messages to A are silently dropped
  1025→// (B sees their own message as "sent" but it never reaches A — we don't reveal
  1026→// to B that they've been blocked, to prevent retaliation).
  1027→//
  1028→// Blocking does NOT delete the existing conversation history — it just stops
  1029→// new messages from flowing.
  1030→// ==========================================================================
  1031→
  1032→model Block {
  1033→  id          String   @id @default(cuid())
  1034→  // The user who initiated the block (the blocker)
  1035→  blockerId   String
  1036→  // The user who is blocked
  1037→  blockedId   String
  1038→  // Optional reason for the block (for the blocker's own record; never shown
  1039→  // to the blocked user)
  1040→  reason       String?
  1041→  createdAt   DateTime @default(now())
  1042→
  1043→  blocker     User     @relation("Blocker", fields: [blockerId], references: [id], onDelete: Cascade)
  1044→  blocked     User     @relation("Blocked", fields: [blockedId], references: [id], onDelete: Cascade)
  1045→
  1046→  @@unique([blockerId, blockedId])
  1047→  @@index([blockerId])
  1048→  @@index([blockedId])
