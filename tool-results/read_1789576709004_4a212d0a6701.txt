    43→// ---- Protection helpers -------------------------------------------------
    44→// Resolves whether a message is effectively protected from screenshot /
    45→// forwarding. The effective flag is:
    46→//   - the explicit `message.protected` value if it is a boolean
    47→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
    48→//     the sender's settings here, so we just use the explicit value — the
    49→//     backend always resolves and persists it on send, so the frontend
    50→//     always sees the resolved boolean).
    51→//
    52→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
    53→// is true only when the current user is the recipient of a protected
    54→// message and has NOT opted into "privacy always allow".
    55→function useProtectionState(message: ChatMessage) {
    56→  const me = useWaslStore((s) => s.user)
    57→  const isOwner = !!me && message.senderId === me.id
    58→  const alwaysAllow = !!me?.privacyAlwaysAllow
    59→  const isProtected = message.protected === true
    60→  const blocked = isProtected && !isOwner && !alwaysAllow
    61→  return { isProtected, isOwner, alwaysAllow, blocked }
    62→}

    64→// Record a screenshot/copy/save attempt to the audit log via the API and
    65→// show a toast. Idempotent — debounced per-message per-kind to avoid
    66→// spamming the server when the user holds down PrintScreen.
    67→const recentAttemptCache = new Map<string, number>()
    68→function recordAttempt(messageId: string, kind: string) {
    69→  const key = `${messageId}:${kind}`
    70→  const now = Date.now()
    71→  if (recentAttemptCache.has(key)) {
    72→    const last = recentAttemptCache.get(key)!
    73→    // Debounce: at most one record per 5 seconds per kind per message.
    74→    if (now - last < 5000) return
    75→  }
    76→  recentAttemptCache.set(key, now)
    77→  // Fire and forget
    78→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
    79→    method: 'POST',
    80→    headers: { 'Content-Type': 'application/json' },
    81→    body: JSON.stringify({ kind }),
    82→  }).catch(() => {})
    83→}

   673→// Voice message playback bubble with a simple play/pause + waveform.
   674→function VoiceMessagePlayer({
   675→  src,
   676→  mine,
   677→  status,
   678→  createdAt,
   679→}: {
   680→  src: string
   681→  mine: boolean
   682→  status: string
   683→  createdAt: string
   684→}) {
   685→  const [playing, setPlaying] = useState(false)
   686→  const [progress, setProgress] = useState(0)
   687→  const [duration, setDuration] = useState(0)
   688→  const audioRef = useRef<HTMLAudioElement | null>(null)
   689→
   690→  function toggle() {
   691→    if (!audioRef.current) {
   692→      audioRef.current = new Audio(src)
   693→      audioRef.current.addEventListener('loadedmetadata', () => {
   694→        setDuration(audioRef.current?.duration || 0)
   695→      })
   696→      audioRef.current.addEventListener('timeupdate', () => {
   697→        setProgress(audioRef.current?.currentTime || 0)
   698→      })
   699→      audioRef.current.addEventListener('ended', () => {
   700→        setPlaying(false)
   701→        setProgress(0)
   702→      })
   703→    }
   704→    if (playing) {
   705→      audioRef.current.pause()
   706→      setPlaying(false)
   707→    } else {
   708→      audioRef.current.play()
   709→      setPlaying(true)
   710→    }
   711→  }
   712→
   713→  function formatTime(sec: number) {
   714→    if (!sec || Number.isNaN(sec)) return '0:00'
   715→    const m = Math.floor(sec / 60)
   716→    const s = Math.floor(sec % 60)
   717→    return `${m}:${String(s).padStart(2, '0')}`
   718→  }
   719→
   720→  const pct = duration > 0 ? (progress / duration) * 100 : 0
   721→  // Deterministic pseudo-waveform bars
   722→  const bars = Array.from({ length: 28 }, (_, i) => {
   723→    const h = 30 + Math.sin(i * 1.3 + src.length) * 20 + Math.cos(i * 2.1) * 15
   724→    return Math.max(10, Math.min(100, Math.abs(h)))
   725→  })
   726→
   727→  return (
   728→    <div className="flex items-center gap-2 min-w-[200px]">
   729→      <button
   730→        type="button"
   731→        onClick={toggle}
   732→        className="w-9 h-9 rounded-full bg-[var(--wasl-green)] text-white flex items-center justify-center shrink-0 hover:opacity-90"
   733→      >
   734→        {playing ? (
   735→          <Pause className="w-4 h-4" />
   736→        ) : (
   737→          <Play className="w-4 h-4 ml-0.5" />
   738→        )}
   739→      </button>
   740→      <div className="flex-1 min-w-0">
   741→        <div className="flex items-center gap-0.5 h-8">
   742→          {bars.map((h, i) => {
   743→            const barProgress = (i / bars.length) * 100
   744→            const active = barProgress < pct
   745→            return (
   746→              <div
   747→                key={i}
   748→                className={cn(
   749→                  'w-0.5 rounded-full transition-colors',
   750→                  active
   751→                    ? 'bg-[var(--wasl-green)]'
   752→                    : mine
   753→                    ? 'bg-foreground/30'
   754→                    : 'bg-foreground/20'
   755→                )}
   756→                style={{ height: `${h}%` }}
   757→              />
   758→            )
   759→          })}
   760→        </div>
   761→        <div className="flex items-center justify-between mt-0.5">
   762→          <span className="text-[10px] text-foreground/60">
   763→            {formatTime(playing ? progress : duration)}
   764→          </span>
   765→          <span className="inline-flex items-center gap-1 text-[10px] text-foreground/50">
   766→            {formatChatTimestamp(createdAt)}
   767→            {mine && <StatusTicks status={status} />}
   768→          </span>
   769→        </div>
   770→      </div>
   771→    </div>
   772→  )
   773→}

   804→function StatusTicks({
   805→  status,
   806→  className,
   807→  onClick,
   808→}: {
   809→  status: string
   810→  className?: string
   811→  onClick?: () => void
   812→}) {
   813→  if (status === 'sent') {
   814→    return <Check className={cn('w-3.5 h-3.5 inline', className)} />
   815→  }
   816→  if (status === 'delivered') {
   817→    return <CheckCheck className={cn('w-3.5 h-3.5 inline', className)} />
   818→  }
   819→  if (status === 'read') {
   820→    // When onClick is provided, the read-ticks become a clickable button that
   821→    // opens the read-receipts dialog (sender-only feature).
   822→    if (onClick) {
   823→      return (
   824→        <button
   825→          type="button"
   826→          onClick={(e) => {
   827→            e.stopPropagation()
   828→            onClick()
   829→          }}
   830→          className="inline-flex items-center hover:scale-110 transition-transform"
   831→          title="Read — click to see who read this message"
   832→          aria-label="Read — click to see details"
   833→        >
   834→          <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />
   835→        </button>
   836→      )
   837→    }
   838→    return (
   839→      <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />
   840→    )
   841→  }
   842→  return <Clock className={cn('w-3.5 h-3.5 inline', className)} />

   845→function groupReactions(reactions: Reaction[]) {
   846→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
   847→  for (const r of reactions) {
   848→    const ex = map.get(r.emoji)
   849→    if (ex) {
   850→      ex.count++
   851→      ex.users.push(r.userId)
   852→    } else {
   853→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
   854→    }
   855→  }
   856→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
   857→}
