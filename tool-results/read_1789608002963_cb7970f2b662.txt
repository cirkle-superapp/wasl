   207→  async function toggleMute() {
   208→    const next = !muted
   209→    setMuted(next) // optimistic
   210→    try {
   211→      const res = await fetch(
   212→        `/api/conversations/${activeConversationId}/mute`,
   213→        {
   214→          method: 'POST',
   215→          headers: { 'Content-Type': 'application/json' },
   216→          body: JSON.stringify({ muted: next }),
   217→        }
   218→      )
   219→      if (!res.ok) {
   220→        setMuted(!next) // rollback
   221→        toast.error('Failed to update mute setting')
   222→        return
   223→      }
   224→      toast.success(next ? 'Notifications muted' : 'Notifications unmuted')
   225→    } catch {
   226→      setMuted(!next) // rollback
   227→      toast.error('Network error')
   228→    }

   248→  // Remove a member from the group (admin only).
   249→  async function handleRemoveMember(userId: string, name: string) {
   250→    if (!activeConversationId) return
   251→    if (
   252→      !confirm(
   253→        `Remove ${name} from this group? They will no longer be able to see or send messages here.`
   254→      )
   255→    )
   256→      return
   257→    setRemovingUserId(userId)
   258→    try {
   259→      const res = await fetch(
   260→        `/api/conversations/${activeConversationId}/members?userId=${encodeURIComponent(userId)}`,
   261→        { method: 'DELETE' }
   262→      )
   263→      if (!res.ok) {
   264→        const data = await res.json().catch(() => null)
   265→        toast.error(data?.error || 'Failed to remove member')
   266→        return
   267→      }
   268→      // Optimistic local update — drop the removed member from the list
   269→      setLocalParticipants((prev) => {
   270→        const base = prev ?? conversation?.participants ?? []
   271→        return base.filter((p) => p.userId !== userId)
   272→      })
   273→      toast.success(`${name} removed`)
   274→      // Refetch so the sidebar member count + last-message preview refresh.
   275→      void refreshConversation()
   276→    } catch {
   277→      toast.error('Network error')
   278→    } finally {
   279→      setRemovingUserId(null)
   280→    }
   281→  }

   942→// AddMemberDialog — search users and add them to the group (admin only).
   943→// Excludes existing members from the search results. Calls
   944→// POST /api/conversations/:id/members with { userId } when a search result
   945→// is clicked.
   946→// ---------------------------------------------------------------------------
   947→
   948→type SearchUser = {
   949→  id: string
   950→  username: string
   951→  name: string
   952→  phone: string | null
   953→  avatar: string | null
   954→  avatarColor: string | null
   955→  about: string
   956→  online: boolean
   957→  lastSeen: string
   958→  verified: boolean
   959→}
   960→
   961→function AddMemberDialog({
   962→  open,
   963→  onOpenChange,
   964→  conversationId,
   965→  currentMemberIds,
   966→  onMemberAdded,
   967→}: {
   968→  open: boolean
   969→  onOpenChange: (v: boolean) => void
   970→  conversationId: string
   971→  currentMemberIds: string[]
   972→  onMemberAdded: (u: SearchUser) => void
   973→}) {
   974→  const [query, setQuery] = useState('')
   975→  const [results, setResults] = useState<SearchUser[]>([])
   976→  const [loading, setLoading] = useState(false)
   977→  const [addingId, setAddingId] = useState<string | null>(null)
   978→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   979→  const memberSet = useRef<Set<string>>(new Set(currentMemberIds))
   980→
   981→  // Keep the current-member Set in sync whenever the prop changes (e.g. after
   982→  // an optimistic add/remove in the parent).
   983→  useEffect(() => {
   984→    memberSet.current = new Set(currentMemberIds)
   985→  }, [currentMemberIds])
   986→
   987→  // Reset search state whenever the dialog closes so it's clean on re-open.
   988→  useEffect(() => {
   989→    if (!open) {
   990→      setQuery('')
   991→      setResults([])
   992→      setAddingId(null)
   993→      if (debounceRef.current) {
   994→        clearTimeout(debounceRef.current)
   995→        debounceRef.current = null
   996→      }
   997→    }
   998→  }, [open])
   999→
  1000→  // Debounced search — fires 300ms after the user stops typing.
  1001→  useEffect(() => {
  1002→    if (!open) return
  1003→    const q = query.trim()
  1004→    if (!q) {
  1005→      setResults([])
  1006→      setLoading(false)
  1007→      return
  1008→    }
  1009→    setLoading(true)
  1010→    if (debounceRef.current) clearTimeout(debounceRef.current)
  1011→    debounceRef.current = setTimeout(async () => {
  1012→      try {
  1013→        const res = await fetch(
  1014→          `/api/users/search?q=${encodeURIComponent(q)}`,
  1015→          { cache: 'no-store' }
  1016→        )
  1017→        if (!res.ok) {
  1018→          setResults([])
  1019→          return
  1020→        }
  1021→        const data = await res.json()
  1022→        setResults((data.users as SearchUser[]) || [])
  1023→      } catch {
  1024→        setResults([])
  1025→      } finally {
  1026→        setLoading(false)
  1027→      }
  1028→    }, 300)
  1029→    return () => {
  1030→      if (debounceRef.current) {
  1031→        clearTimeout(debounceRef.current)
  1032→        debounceRef.current = null
  1033→      }
  1034→    }
  1035→  }, [query, open])
  1036→
  1037→  async function handleAdd(user: SearchUser) {
  1038→    setAddingId(user.id)
  1039→    try {
  1040→      const res = await fetch(
  1041→        `/api/conversations/${conversationId}/members`,
  1042→        {
  1043→          method: 'POST',
  1044→          headers: { 'Content-Type': 'application/json' },
  1045→          body: JSON.stringify({ userId: user.id }),
  1046→        }
  1047→      )
  1048→      if (!res.ok) {
  1049→        const data = await res.json().catch(() => null)
  1050→        toast.error(data?.error || 'Failed to add member')
  1051→        return
  1052→      }
  1053→      toast.success(`${user.name} added to the group`)
  1054→      onMemberAdded(user)
  1055→      // Remove from the visible search results so the user can't add twice.
  1056→      setResults((prev) => prev.filter((u) => u.id !== user.id))
  1057→      // Close after a successful add.
  1058→      onOpenChange(false)
  1059→    } catch {
  1060→      toast.error('Network error')
  1061→    } finally {
  1062→      setAddingId(null)
  1063→    }
  1064→  }
  1065→
  1066→  return (
  1067→    <Dialog open={open} onOpenChange={onOpenChange}>
  1068→      <DialogContent className="max-w-md max-h-[80vh] flex flex-col p-0 gap-0">
  1069→        <DialogHeader className="px-4 pt-4 pb-2">
  1070→          <DialogTitle>Add member</DialogTitle>
  1071→          <DialogDescription>
  1072→            Search for a Wasl user by name, username, or phone number. They
  1073→            will be added to this group as a regular member.
  1074→          </DialogDescription>
  1075→        </DialogHeader>
  1076→        <div className="px-4 pb-2">
  1077→          <div className="relative">
  1078→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground pointer-events-none" />
  1079→            <Input
  1080→              autoFocus
  1081→              value={query}
  1082→              onChange={(e) => setQuery(e.target.value)}
  1083→              placeholder="Search name, username, or phone…"
  1084→              className="pl-9"
  1085→            />
  1086→            {loading && (
  1087→              <Loader2 className="absolute right-2.5 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />
  1088→            )}
  1089→          </div>
  1090→        </div>
  1091→        <div className="flex-1 overflow-y-auto wasl-scroll px-2 pb-2">
  1092→          {!loading && query.trim() === '' ? (
  1093→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
  1094→              Start typing to search for someone to add.
  1095→            </div>
  1096→          ) : !loading && results.length === 0 ? (
  1097→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
  1098→              No users found for &ldquo;{query}&rdquo;.
  1099→            </div>
  1100→          ) : (
  1101→            <div className="space-y-0.5">
  1102→              {results.map((u) => {
  1103→                const alreadyMember = memberSet.current.has(u.id)
  1104→                return (
  1105→                  <button
  1106→                    key={u.id}
  1107→                    type="button"
  1108→                    disabled={alreadyMember || addingId === u.id}
  1109→                    onClick={() => handleAdd(u)}
  1110→                    className={cn(
  1111→                      'w-full flex items-center gap-3 p-2 rounded-lg text-left transition-colors',
  1112→                      alreadyMember
  1113→                        ? 'opacity-50 cursor-not-allowed'
  1114→                        : 'hover:bg-muted/60 cursor-pointer'
  1115→                    )}
  1116→                  >
  1117→                    <WaslAvatar
  1118→                      name={u.name}
  1119→                      src={u.avatar}
  1120→                      color={u.avatarColor}
  1121→                      size={36}
  1122→                      online={u.online}
  1123→                      showStatus
  1124→                    />
  1125→                    <div className="flex-1 min-w-0">
  1126→                      <div className="text-sm font-medium truncate">{u.name}</div>
  1127→                      <div className="text-xs text-muted-foreground truncate">
  1128→                        @{u.username}
  1129→                        {u.phone ? ` · ${u.phone}` : ''}
  1130→                      </div>
  1131→                    </div>
  1132→                    {alreadyMember ? (
  1133→                      <span className="text-[10px] text-muted-foreground shrink-0">
  1134→                        Already in group
  1135→                      </span>
  1136→                    ) : addingId === u.id ? (
  1137→                      <Loader2 className="w-4 h-4 animate-spin text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1138→                    ) : (
  1139→                      <UserPlus className="w-4 h-4 text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1140→                    )}
  1141→                  </button>
  1142→                )
  1143→              })}
  1144→            </div>
  1145→          )}
  1146→        </div>
  1147→      </DialogContent>
  1148→    </Dialog>
  1149→  )

  1157→function RenameGroupDialog({
  1158→  open,

  1163→}: {

  1165→  onOpenChange: (v: boolean) => void

  1268→  open: boolean

  1275→  const [saving, setSaving] = useState(false)
  1276→

  1278→  // the canonical description (not a stale draft from a previous open).
  1279→  useEffect(() => {
  1280→    if (open) setValue(currentDescription ?? '')
  1281→  }, [open, currentDescription])
  1282→
  1283→  const trimmed = value.trim()
  1284→  // Valid as long as it's within the 500-char limit. Empty is allowed (clears).
  1285→  const valid = value.length <= 500

  1287→  const previousTrimmed = (currentDescription ?? '').trim()

  1289→
  1290→  async function handleSave() {
  1291→    if (!valid || !changed) return
  1292→    setSaving(true)

  1294→      const res = await fetch(`/api/conversations/${conversationId}`, {
  1295→        method: 'PATCH',

  1297→        body: JSON.stringify({ description: value }),
  1298→      })

  1300→        const data = await res.json().catch(() => null)
  1301→        toast.error(data?.error || 'Failed to update description')
  1302→        return
  1303→      }

  1305→      toast.success(
  1306→        next ? 'Description updated' : 'Description deleted'
  1307→      )
  1308→      onSaved(next)
  1309→      onOpenChange(false)
  1310→    } catch {
  1311→      toast.error('Network error')
  1312→    } finally {
  1313→      setSaving(false)
  1314→    }

  1326→            preserved. Leave empty to clear the description.
  1327→          </DialogDescription>
  1328→        </DialogHeader>
  1329→        <div className="space-y-2">
  1330→          <Label htmlFor="wasl-group-description-input">Description</Label>
  1331→          <Textarea
  1332→            id="wasl-group-description-input"
  1333→            autoFocus

  1335→            onChange={(e) => setValue(e.target.value)}

  1337→            placeholder="What is this group about?"
  1338→            className="min-h-24 resize-y"
