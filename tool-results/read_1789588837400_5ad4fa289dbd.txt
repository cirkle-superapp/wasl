   295→        }
   296→      } catch {
   297→        // ignore
   298→      }
   299→    }

   358→  useEffect(() => {
   359→    function onJumpToMessage(e: Event) {
   360→      const messageId = (e as CustomEvent<string>).detail

   362→      // Prefer the scroll container scope, but fall back to a document-wide

   370→          `[data-message-id="${messageId}"]`

   397→    }

   542→  // ---- Message actions: react / star / copy / delete ------------------------

   544→    async (messageId: string, emoji: string) => {

   547→      const msgs = messagesByConversation[activeConversationId] || []
   548→      const existing = msgs
   549→        .find((m) => m.id === messageId)
   550→        ?.reactions?.find((r) => r.userId === user.id)
   551→      // If the user already reacted with the same emoji → remove; else set.
   552→      if (existing && existing.emoji === emoji) {
   553→        toggleReaction(activeConversationId, messageId, null)
   554→      } else {
   555→        toggleReaction(activeConversationId, messageId, {
   556→          id: existing?.id,
   557→          userId: user.id,
   558→          emoji,
   559→        })
   560→      }
   561→      try {
   562→        await fetch(`/api/messages/${messageId}/reactions`, {
   563→          method: 'POST',
   564→          headers: { 'Content-Type': 'application/json' },
   565→          body: JSON.stringify({ emoji }),
   566→        })
   567→        getSocket().emit('message:reacted', {
   568→          conversationId: activeConversationId,
   569→          messageId,
   570→        })
   571→      } catch (e) {
   572→        console.error(e)
   573→      }
   574→    },
   575→    [activeConversationId, user?.id, messagesByConversation, toggleReaction]

   578→  const handleStar = useCallback(
   579→    async (messageId: string) => {
   580→      if (!activeConversationId) return
   581→      const msgs = messagesByConversation[activeConversationId] || []
   582→      const currentlyStarred = msgs.find((m) => m.id === messageId)?.starred
   583→      setStarred(activeConversationId, messageId, !currentlyStarred)
