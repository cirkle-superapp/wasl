    47→// ---- Search highlight helper ---------------------------------------------
    48→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    49→// match in a <mark> with a visible yellow background so the user can see why
    50→// the message was returned by the chat-search dialog.
    51→//
    52→// Supports multi-word queries: if the query contains spaces, each word is
    53→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    54→// and "Wasl" separately). Each word is regex-escaped so special characters
    55→// (`(`, `.`, `*`, …) are matched literally.
    56→//
    57→// Returns the original `text` (a single string node) when either the text or
    58→// query is empty, so plain messages with no active query render with zero
    59→// overhead and no behavioural change vs. before.
    60→function highlightText(text: string, query: string): React.ReactNode {
    61→  if (!text) return text
    62→  const q = query.trim()
    63→  if (!q) return text
    64→
    65→  // Split the query into individual words for multi-word highlighting.
    66→  // Filter out empty strings from multiple consecutive spaces.
    67→  const words = q.split(/\s+/).filter(Boolean)
    68→  if (words.length === 0) return text
    69→
    70→  // Build a single regex that matches ANY of the words (OR).
    71→  // Each word is regex-escaped so special characters are literal.
    72→  const escapedWords = words.map((w) =>
    73→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    74→  )
    75→  const pattern = escapedWords.join('|')
    76→  let regex: RegExp
    77→  try {
    78→    regex = new RegExp(`(${pattern})`, 'gi')
    79→  } catch {
    80→    // Malformed query — bail to plain text rather than crashing the bubble.
    81→    return text
    82→  }
    83→  // String.split with a capturing group keeps the matched delimiters in the
    84→  // result array, so odd indices are the query matches (highlight them) and
    85→  // even indices are the surrounding plain text.
    86→  const parts = text.split(regex)
    87→  if (parts.length === 1) {
    88→    // No matches — render as a single plain string node.
    89→    return text
    90→  }
    91→  return parts.map((part, i) =>
    92→    i % 2 === 1 ? (
    93→      <mark
    94→        key={i}
    95→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    96→      >
    97→        {part}
    98→      </mark>
    99→    ) : (
   100→      // Render empty strings as `null` so React doesn't complain about
   101→      // adjacent text separators. Non-empty even-indexed parts become plain
   102→      // string nodes (no wrapping element needed).
   103→      part || null
   104→    )
   105→  )
   106→}

   108→// ---- File-attachment content parser --------------------------------------
   109→// PDF / document / audio messages store their payload as a small JSON blob
   110→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   111→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   112→// resilient when the JSON shape isn't present.
   113→export function parseFileContent(
   114→  content: string
   115→): { url: string; name?: string; size?: number } | null {
   116→  if (!content) return null
   117→  try {
   118→    const parsed = JSON.parse(content)
   119→    if (
   120→      parsed &&
   121→      typeof parsed === 'object' &&
   122→      typeof parsed.url === 'string'
   123→    ) {
   124→      return {
   125→        url: parsed.url,
   126→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   127→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   128→      }
   129→    }
   130→  } catch {
   131→    // Not JSON — fall through to plain-URL detection.
   132→  }
   133→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   134→  if (content.startsWith('http') || content.startsWith('/')) {
   135→    return { url: content }
   136→  }
   137→  return null
   138→}

   140→export function formatFileSize(bytes: number): string {
   141→  if (bytes < 1024) return `${bytes} B`
   142→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   143→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   144→}

   146→// Derive a filename from a URL when the parsed payload has no `name`.
   147→function deriveFilenameFromUrl(url: string, fallback: string): string {
   148→  try {
   149→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   150→    const last = u.pathname.split('/').filter(Boolean).pop()
   151→    return last ? decodeURIComponent(last) : fallback
   152→  } catch {
   153→    return fallback
   154→  }

   157→// ---- Protection helpers -------------------------------------------------
   158→// Resolves whether a message is effectively protected from screenshot /
   159→// forwarding. The effective flag is:
   160→//   - the explicit `message.protected` value if it is a boolean
   161→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   162→//     the sender's settings here, so we just use the explicit value — the
   163→//     backend always resolves and persists it on send, so the frontend
   164→//     always sees the resolved boolean).
   165→//
   166→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   167→// is true only when the current user is the recipient of a protected
   168→// message and has NOT opted into "privacy always allow".
   169→function useProtectionState(message: ChatMessage) {
   170→  const me = useWaslStore((s) => s.user)
   171→  const isOwner = !!me && message.senderId === me.id
   172→  const alwaysAllow = !!me?.privacyAlwaysAllow
   173→  const isProtected = message.protected === true
   174→  const blocked = isProtected && !isOwner && !alwaysAllow
   175→  return { isProtected, isOwner, alwaysAllow, blocked }
   176→}

   178→// Record a screenshot/copy/save attempt to the audit log via the API and
   179→// show a toast. Idempotent — debounced per-message per-kind to avoid
   180→// spamming the server when the user holds down PrintScreen.
   181→const recentAttemptCache = new Map<string, number>()
   182→function recordAttempt(messageId: string, kind: string) {
   183→  const key = `${messageId}:${kind}`
   184→  const now = Date.now()
   185→  if (recentAttemptCache.has(key)) {
   186→    const last = recentAttemptCache.get(key)!
   187→    // Debounce: at most one record per 5 seconds per kind per message.
   188→    if (now - last < 5000) return
   189→  }
   190→  recentAttemptCache.set(key, now)
   191→  // Fire and forget
   192→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   193→    method: 'POST',
   194→    headers: { 'Content-Type': 'application/json' },
   195→    body: JSON.stringify({ kind }),
   196→  }).catch(() => {})
   197→}

  1020→// ---- File-attachment card (PDF / document) -------------------------------
  1021→// Renders a compact "file card" inside the bubble: a coloured FileText icon
  1022→// (red for PDF, teal/green for documents), the original filename + human-
  1023→// readable size, and a Download button that opens the URL in a new tab.
  1024→//
  1025→// `blocked` is set when the message is protected AND the current user is the
  1026→// recipient — in that case the Download button is disabled and shows the
  1027→// protected warning toast instead of opening the link.
  1028→function PdfDocumentCardContent({
  1029→  message,
  1030→  mine,
  1031→  blocked,
  1032→  isProtected,
  1033→  onOpenReadReceipts,
  1034→}: {
  1035→  message: ChatMessage
  1036→  mine: boolean
  1037→  blocked: boolean
  1038→  isProtected: boolean
  1039→  onOpenReadReceipts: () => void
  1040→}) {
  1041→  const parsed = parseFileContent(message.content)
  1042→  const isPdf = message.type === 'pdf'
  1043→  const fallbackName = isPdf ? 'PDF document' : 'Document'
  1044→  const name =
  1045→    parsed?.name ||
  1046→    (parsed?.url ? deriveFilenameFromUrl(parsed.url, fallbackName) : fallbackName)
  1047→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : isPdf ? 'PDF' : 'Document'
  1048→
  1049→  // PDF → red icon. Document → teal/green icon (uses wasl theme tokens so it
  1050→  // respects light/dark mode and the Cirkle gold theme). No indigo/blue.
  1051→  const iconColor = isPdf
  1052→    ? 'text-red-500'
  1053→    : 'text-[var(--wasl-teal)] dark:text-[var(--wasl-green)]'
  1054→  const iconBg = isPdf
  1055→    ? 'bg-red-500/10'
  1056→    : 'bg-[var(--wasl-teal)]/10 dark:bg-[var(--wasl-green)]/10'
  1057→
  1058→  return (
  1059→    <div className="rounded-lg overflow-hidden max-w-[280px]">
  1060→      <div className="flex items-center gap-3 p-2 min-w-[220px]">
  1061→        <div className={cn('w-10 h-10 rounded-lg flex items-center justify-center shrink-0', iconBg)}>
  1062→          <FileText className={cn('w-5 h-5', iconColor)} />
  1063→        </div>
  1064→        <div className="flex-1 min-w-0">
  1065→          <div className="text-sm font-medium truncate text-foreground" title={name}>
  1066→            {name}
  1067→          </div>
  1068→          <div className="text-[11px] text-foreground/60 uppercase tracking-wide">
  1069→            {sizeLabel}
  1070→          </div>
  1071→        </div>
  1072→        {blocked ? (
  1073→          <button
  1074→            type="button"
  1075→            onClick={(e) => {
  1076→              e.stopPropagation()
  1077→              toast.error(PROTECTED_WARNING, { duration: 4000 })

  1113→// ---- Audio URL card ------------------------------------------------------

  1241→function StatusTicks({

  1282→function groupReactions(reactions: Reaction[]) {
  1283→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1284→  for (const r of reactions) {
  1285→    const ex = map.get(r.emoji)
  1286→    if (ex) {
  1287→      ex.count++
  1288→      ex.users.push(r.userId)
  1289→    } else {
  1290→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1291→    }
  1292→  }
  1293→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
  1294→}
