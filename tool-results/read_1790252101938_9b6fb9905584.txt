    72→  // Live username availability check (debounced)
    73→  useEffect(() => {
    74→    if (mode !== 'signup' || !usernameTouched) return
    75→    const u = username.trim().toLowerCase()
    76→    if (!u) {
    77→      setUsernameStatus({ checking: false, available: null, message: '', suggestions: [] })

    84→        const res = await fetch(`/api/auth/check-username?u=${encodeURIComponent(u)}`)
    85→        const data = await res.json()
    86→        setUsernameStatus({ checking: false, available: data.available, message: data.message || '', suggestions: data.suggestions || [] })
    87→      } catch {
    88→        setUsernameStatus({ checking: false, available: null, message: '', suggestions: [] })
    89→      }
    90→    }, 350)
    91→    return () => { if (usernameTimer.current) clearTimeout(usernameTimer.current) }
    92→  }, [username, usernameTouched, mode])

    94→  // Auto-generate username from name

    96→    if (mode !== 'signup' || usernameTouched) return
    97→    const auto = name.trim().toLowerCase().replace(/[^a-z0-9_]/g, '_').replace(/_+/g, '_').slice(0, 15)
    98→    if (auto && auto !== username) setUsername(auto)

   101→  const authenticate = useCallback(
   102→    async (
   103→      overrideMode?: 'login' | 'signup',
   104→      overrideIdentifier?: string,
   105→      overridePassword?: string,
   106→      overrideName?: string,
   107→      overrideUsername?: string,
   108→      overrideSignupId?: string
   109→    ) => {
   110→      const m = overrideMode ?? mode
   111→      const p = overridePassword ?? password
   112→      setLoading(true)
   113→      try {
   114→        if (m === 'login') {
   115→          const id = overrideIdentifier ?? identifier
   116→          if (!id.trim()) { toast.error('Please enter your email, phone, or username'); return }
   117→          if (!p) { toast.error('Please enter your password'); return }
   118→          const res = await fetch('/api/auth/login', {
   119→            method: 'POST',
   120→            headers: { 'Content-Type': 'application/json' },
   121→            body: JSON.stringify({ identifier: id, password: p }),
   122→          })
   123→          const data = await res.json()
   124→          if (!res.ok) { toast.error(data?.error || 'Login failed'); return }
   125→          setUser(data)
   126→          toast.success(`Welcome to Wasl, ${data.name}!`)
   127→          router.refresh()
   128→        } else {
   129→          const n = (overrideName ?? name).trim()
   130→          const u = (overrideUsername ?? username).trim().toLowerCase()
   131→          const sid = overrideSignupId ?? signupIdentifier
   132→          if (!n || n.length < 2) { toast.error('Please enter your name (at least 2 characters)'); return }
   133→          if (!u || u.length < 3) { toast.error('Please choose a Cirkle username (at least 3 characters)'); return }
   134→          if (!p || p.length < 6) { toast.error('Password must be at least 6 characters'); return }
   135→          const res = await fetch('/api/auth/signup', {
   136→            method: 'POST',
   137→            headers: { 'Content-Type': 'application/json' },
   138→            body: JSON.stringify({ identifier: sid || undefined, password: p, name: n, username: u }),
   139→          })
   140→          const data = await res.json()
   141→          if (!res.ok) {
   142→            if (data.suggestions?.length) { toast.error(data.error, { description: `Try: ${data.suggestions.join(', ')}` }) }
   143→            else { toast.error(data?.error || 'Sign up failed') }
   144→            return
   145→          }
   146→          setUser(data)
   147→          toast.success(`Welcome to Wasl, ${data.name}!`)
   148→          router.refresh()
   149→        }
   150→      } catch (err) {
   151→        console.error(err)
   152→        toast.error('Network error')
   153→      } finally {
   154→        setLoading(false)
   155→      }
   156→    },
