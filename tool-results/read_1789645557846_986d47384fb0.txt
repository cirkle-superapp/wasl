    47→// ---- Search highlight helper ---------------------------------------------
    48→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    49→// match in a <mark> with a visible yellow background so the user can see why
    50→// the message was returned by the chat-search dialog.
    51→//
    52→// Supports multi-word queries: if the query contains spaces, each word is
    53→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    54→// and "Wasl" separately). Each word is regex-escaped so special characters
    55→// (`(`, `.`, `*`, …) are matched literally.
    56→//
    57→// Returns the original `text` (a single string node) when either the text or
    58→// query is empty, so plain messages with no active query render with zero
    59→// overhead and no behavioural change vs. before.
    60→function highlightText(text: string, query: string): React.ReactNode {
    61→  if (!text) return text
    62→  const q = query.trim()
    63→  if (!q) return text
    64→
    65→  // Split the query into individual words for multi-word highlighting.
    66→  // Filter out empty strings from multiple consecutive spaces.
    67→  const words = q.split(/\s+/).filter(Boolean)
    68→  if (words.length === 0) return text
    69→
    70→  // Build a single regex that matches ANY of the words (OR).
    71→  // Each word is regex-escaped so special characters are literal.
    72→  const escapedWords = words.map((w) =>
    73→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    74→  )
    75→  const pattern = escapedWords.join('|')
    76→  let regex: RegExp
    77→  try {
    78→    regex = new RegExp(`(${pattern})`, 'gi')
    79→  } catch {
    80→    // Malformed query — bail to plain text rather than crashing the bubble.
    81→    return text
    82→  }
    83→  // String.split with a capturing group keeps the matched delimiters in the
    84→  // result array, so odd indices are the query matches (highlight them) and
    85→  // even indices are the surrounding plain text.
    86→  const parts = text.split(regex)
    87→  if (parts.length === 1) {
    88→    // No matches — render as a single plain string node.
    89→    return text
    90→  }
    91→  return parts.map((part, i) =>
    92→    i % 2 === 1 ? (
    93→      <mark
    94→        key={i}
    95→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    96→      >
    97→        {part}
    98→      </mark>
    99→    ) : (
   100→      // Render empty strings as `null` so React doesn't complain about
   101→      // adjacent text separators. Non-empty even-indexed parts become plain
   102→      // string nodes (no wrapping element needed).
   103→      part || null
   104→    )
   105→  )
   106→}

   108→// ---- File-attachment content parser --------------------------------------
   109→// PDF / document / audio messages store their payload as a small JSON blob
   110→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   111→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   112→// resilient when the JSON shape isn't present.
   113→export function parseFileContent(
   114→  content: string
   115→): { url: string; name?: string; size?: number } | null {
   116→  if (!content) return null
   117→  try {
   118→    const parsed = JSON.parse(content)
   119→    if (
   120→      parsed &&
   121→      typeof parsed === 'object' &&
   122→      typeof parsed.url === 'string'
   123→    ) {
   124→      return {
   125→        url: parsed.url,
   126→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   127→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   128→      }
   129→    }
   130→  } catch {
   131→    // Not JSON — fall through to plain-URL detection.
   132→  }
   133→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   134→  if (content.startsWith('http') || content.startsWith('/')) {
   135→    return { url: content }
   136→  }
   137→  return null
   138→}

   140→export function formatFileSize(bytes: number): string {
   141→  if (bytes < 1024) return `${bytes} B`
   142→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   143→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   144→}

   146→// Derive a filename from a URL when the parsed payload has no `name`.
   147→function deriveFilenameFromUrl(url: string, fallback: string): string {
   148→  try {
   149→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   150→    const last = u.pathname.split('/').filter(Boolean).pop()
   151→    return last ? decodeURIComponent(last) : fallback
   152→  } catch {
   153→    return fallback
   154→  }

   157→// ---- Protection helpers -------------------------------------------------
   158→// Resolves whether a message is effectively protected from screenshot /
   159→// forwarding. The effective flag is:
   160→//   - the explicit `message.protected` value if it is a boolean
   161→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   162→//     the sender's settings here, so we just use the explicit value — the
   163→//     backend always resolves and persists it on send, so the frontend
   164→//     always sees the resolved boolean).
   165→//
   166→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   167→// is true only when the current user is the recipient of a protected
   168→// message and has NOT opted into "privacy always allow".
   169→function useProtectionState(message: ChatMessage) {
   170→  const me = useWaslStore((s) => s.user)
   171→  const isOwner = !!me && message.senderId === me.id
   172→  const alwaysAllow = !!me?.privacyAlwaysAllow
   173→  const isProtected = message.protected === true
   174→  const blocked = isProtected && !isOwner && !alwaysAllow
   175→  return { isProtected, isOwner, alwaysAllow, blocked }
   176→}

   178→// Record a screenshot/copy/save attempt to the audit log via the API and
   179→// show a toast. Idempotent — debounced per-message per-kind to avoid
   180→// spamming the server when the user holds down PrintScreen.
   181→const recentAttemptCache = new Map<string, number>()
   182→function recordAttempt(messageId: string, kind: string) {
   183→  const key = `${messageId}:${kind}`
   184→  const now = Date.now()
   185→  if (recentAttemptCache.has(key)) {
   186→    const last = recentAttemptCache.get(key)!
   187→    // Debounce: at most one record per 5 seconds per kind per message.
   188→    if (now - last < 5000) return
   189→  }
   190→  recentAttemptCache.set(key, now)
   191→  // Fire and forget
   192→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   193→    method: 'POST',
   194→    headers: { 'Content-Type': 'application/json' },
   195→    body: JSON.stringify({ kind }),
   196→  }).catch(() => {})
   197→}

  1061→// ---- File-attachment card (PDF / document) -------------------------------
  1062→// Renders a compact "file card" inside the bubble: a coloured FileText icon
  1063→// (red for PDF, teal/green for documents), the original filename + human-
  1064→// readable size, and a Download button that opens the URL in a new tab.
  1065→//
  1066→// `blocked` is set when the message is protected AND the current user is the
  1067→// recipient — in that case the Download button is disabled and shows the
  1068→// protected warning toast instead of opening the link.
  1069→function PdfDocumentCardContent({
  1070→  message,
  1071→  mine,
  1072→  blocked,
  1073→  isProtected,
  1074→  onOpenReadReceipts,
  1075→}: {
  1076→  message: ChatMessage
  1077→  mine: boolean
  1078→  blocked: boolean
  1079→  isProtected: boolean
  1080→  onOpenReadReceipts: () => void
  1081→}) {
  1082→  const parsed = parseFileContent(message.content)
  1083→  const isPdf = message.type === 'pdf'
  1084→  const fallbackName = isPdf ? 'PDF document' : 'Document'
  1085→  const name =
  1086→    parsed?.name ||
  1087→    (parsed?.url ? deriveFilenameFromUrl(parsed.url, fallbackName) : fallbackName)
  1088→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : isPdf ? 'PDF' : 'Document'

  1300→function StatusTicks({
  1301→  status,
  1302→  className,
  1303→  onClick,
  1304→  readByEveryone,
  1305→  readCount,
  1306→  totalRecipients,
  1307→}: {
  1308→  status: string
  1309→  className?: string
  1310→  onClick?: () => void
  1311→  // ---- Read-receipt summary (sender-only) -------------------------------
  1312→  // When `readByEveryone` is true, ALL other participants have read the
  1313→  // message — the ticks turn wasl-teal/green, get a subtle "pop" pulse
  1314→  // animation, and a small "Read by all" label appears next to them.
  1315→  // When `readCount > 0` but `readByEveryone` is false, the ticks turn
  1316→  // wasl-teal/green and a small "N/M" count badge appears (e.g., "3/5").
  1317→  // When `readCount === 0` (or undefined), the ticks render in the legacy
  1318→  // muted color (delivered / sent).
  1319→  readByEveryone?: boolean
  1320→  readCount?: number
  1321→  totalRecipients?: number
  1322→}) {
  1323→  // Determine the read state. `readByEveryone` is the richest signal — when
  1324→  // true, we know the message has been read by every recipient. The partial
  1325→  // state (some readers, not all) is shown when readCount > 0 but readByEveryone
  1326→  // is false. We fall back to the legacy `status === 'read'` path for
  1327→  // 1-on-1 chats or older messages where the summary isn't populated.
  1328→  const hasReadCount =
  1329→    typeof readCount === 'number' &&
  1330→    typeof totalRecipients === 'number' &&
  1331→    totalRecipients > 0
  1332→  const isReadAll = readByEveryone === true
  1333→  const isPartialRead = !isReadAll && hasReadCount && readCount! > 0
  1334→  const isRead = isReadAll || isPartialRead || status === 'read'
  1335→
  1336→  // Single-check for freshly sent (no delivery confirmation yet).
  1337→  if (status === 'sent' && !isRead) {
  1338→    return <Check className={cn('w-3.5 h-3.5 inline', className)} />
  1339→  }
  1340→  // Pending clock for unknown / pending states.
  1341→  if (!isRead && status !== 'delivered') {
  1342→    return <Clock className={cn('w-3.5 h-3.5 inline', className)} />
  1343→  }
  1344→
  1345→  // Resolve tooltip text + tick color for the double-check state.
  1346→  let title: string
  1347→  let tickColorClass: string
  1348→  let label: string | null = null
  1349→  if (isReadAll) {
  1350→    title = 'Read by all'
  1351→    tickColorClass =
  1352→      'text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] wasl-ticks-read-all'
  1353→    label = 'Read by all'
  1354→  } else if (isPartialRead) {
  1355→    title = `Read by ${readCount} of ${totalRecipients}`
  1356→    tickColorClass = 'text-[var(--wasl-teal)] dark:text-[var(--wasl-green)]'
  1357→    label = `${readCount}/${totalRecipients}`
  1358→  } else if (status === 'read') {
  1359→    // Legacy "read" status without a readCount summary (1-on-1 chats where
  1360→    // the server didn't populate the summary, or older messages). Per the
  1361→    // design rules, we use the wasl-teal/green palette — never blue/indigo.
  1362→    title = 'Read — click to see details'
  1363→    tickColorClass = 'text-[var(--wasl-teal)] dark:text-[var(--wasl-green)]'
  1364→  } else {
  1365→    // Delivered (no reads yet).
  1366→    title = 'Delivered'
  1367→    tickColorClass = 'text-foreground/50'
  1368→  }
  1369→
  1370→  const ticksEl = (
  1371→    <CheckCheck className={cn('w-3.5 h-3.5 inline', tickColorClass, className)} />
  1372→  )
  1373→
  1374→  // Optional small muted label that appears next to the ticks. For "Read by
  1375→  // all" we show the full label; for partial reads we show a compact "N/M"
  1376→  // count badge. Both are intentionally tiny (text-[9px]) and muted so they
  1377→  // stay unobtrusive next to the timestamp.
  1378→  const labelEl = label ? (
  1379→    <span className="text-[9px] font-medium leading-none text-foreground/50 ml-0.5 select-none">
  1380→      {label}
  1381→    </span>
  1382→  ) : null
  1383→
  1384→  // When onClick is provided, the ticks become a clickable button that opens
  1385→  // the read-receipts dialog (sender-only feature).
  1386→  if (onClick) {
  1387→    return (
  1388→      <button
  1389→        type="button"
  1390→        onClick={(e) => {
  1391→          e.stopPropagation()
  1392→          onClick()
  1393→        }}
  1394→        className="inline-flex items-center hover:scale-110 transition-transform"
  1395→        title={title}
  1396→        aria-label={title}
  1397→      >
  1398→        {ticksEl}
  1399→        {labelEl}
  1400→      </button>
  1401→    )
  1402→  }
  1403→  return (
  1404→    <span className="inline-flex items-center" title={title}>
  1405→      {ticksEl}
  1406→      {labelEl}
  1407→    </span>
  1408→  )
  1409→}

  1411→function groupReactions(reactions: Reaction[]) {
  1412→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1413→  for (const r of reactions) {
  1414→    const ex = map.get(r.emoji)
  1415→    if (ex) {
  1416→      ex.count++
  1417→      ex.users.push(r.userId)
  1418→    } else {
  1419→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1420→    }
  1421→  }
  1422→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
  1423→}
