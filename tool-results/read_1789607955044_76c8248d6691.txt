    46→    46→// ---- Search highlight helper ---------------------------------------------
    47→    47→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    48→    48→// match in a <mark> with a visible yellow background so the user can see why
    49→    49→// the message was returned by the chat-search dialog.
    50→    50→//
    51→    51→// Supports multi-word queries: if the query contains spaces, each word is
    52→    52→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    53→    53→// and "Wasl" separately). Each word is regex-escaped so special characters
    54→    54→// (`(`, `.`, `*`, …) are matched literally.
    55→    55→//
    56→    56→// Returns the original `text` (a single string node) when either the text or
    57→    57→// query is empty, so plain messages with no active query render with zero
    58→    58→// overhead and no behavioural change vs. before.
    59→    59→function highlightText(text: string, query: string): React.ReactNode {
    60→    60→  if (!text) return text
    61→    61→  const q = query.trim()
    62→    62→  if (!q) return text
    63→    63→
    64→    64→  // Split the query into individual words for multi-word highlighting.
    65→    65→  // Filter out empty strings from multiple consecutive spaces.
    66→    66→  const words = q.split(/\s+/).filter(Boolean)
    67→    67→  if (words.length === 0) return text
    68→    68→
    69→    69→  // Build a single regex that matches ANY of the words (OR).
    70→    70→  // Each word is regex-escaped so special characters are literal.
    71→    71→  const escapedWords = words.map((w) =>
    72→    72→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    73→    73→  )
    74→    74→  const pattern = escapedWords.join('|')
    75→    75→  let regex: RegExp
    76→    76→  try {
    77→    77→    regex = new RegExp(`(${pattern})`, 'gi')
    78→    78→  } catch {
    79→    79→    // Malformed query — bail to plain text rather than crashing the bubble.
    80→    80→    return text
    81→    81→  }
    82→    82→  // String.split with a capturing group keeps the matched delimiters in the
    83→    83→  // result array, so odd indices are the query matches (highlight them) and
    84→    84→  // even indices are the surrounding plain text.
    85→    85→  const parts = text.split(regex)
    86→    86→  if (parts.length === 1) {
    87→    87→    // No matches — render as a single plain string node.
    88→    88→    return text
    89→    89→  }
    90→    90→  return parts.map((part, i) =>
    91→    91→    i % 2 === 1 ? (
    92→    92→      <mark
    93→    93→        key={i}
    94→    94→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    95→    95→      >
    96→    96→        {part}
    97→    97→      </mark>
    98→    98→    ) : (
    99→    99→      // Render empty strings as `null` so React doesn't complain about
   100→   100→      // adjacent text separators. Non-empty even-indexed parts become plain
   101→   101→      // string nodes (no wrapping element needed).
   102→   102→      part || null
   103→   103→    )
   104→   104→  )
   105→   105→}

   107→   107→// ---- File-attachment content parser --------------------------------------
   108→   108→// PDF / document / audio messages store their payload as a small JSON blob
   109→   109→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   110→   110→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   111→   111→// resilient when the JSON shape isn't present.
   112→   112→export function parseFileContent(
   113→   113→  content: string
   114→   114→): { url: string; name?: string; size?: number } | null {
   115→   115→  if (!content) return null
   116→   116→  try {
   117→   117→    const parsed = JSON.parse(content)
   118→   118→    if (
   119→   119→      parsed &&
   120→   120→      typeof parsed === 'object' &&
   121→   121→      typeof parsed.url === 'string'
   122→   122→    ) {
   123→   123→      return {
   124→   124→        url: parsed.url,
   125→   125→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   126→   126→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   127→   127→      }
   128→   128→    }
   129→   129→  } catch {
   130→   130→    // Not JSON — fall through to plain-URL detection.
   131→   131→  }
   132→   132→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   133→   133→  if (content.startsWith('http') || content.startsWith('/')) {
   134→   134→    return { url: content }
   135→   135→  }
   136→   136→  return null
   137→   137→}

   139→   139→export function formatFileSize(bytes: number): string {
   140→   140→  if (bytes < 1024) return `${bytes} B`
   141→   141→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   142→   142→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   143→   143→}

   145→   145→// Derive a filename from a URL when the parsed payload has no `name`.
   146→   146→function deriveFilenameFromUrl(url: string, fallback: string): string {
   147→   147→  try {
   148→   148→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   149→   149→    const last = u.pathname.split('/').filter(Boolean).pop()
   150→   150→    return last ? decodeURIComponent(last) : fallback
   151→   151→  } catch {
   152→   152→    return fallback
   153→   153→  }
   154→   154→}

   156→   156→// ---- Protection helpers -------------------------------------------------
   157→   157→// Resolves whether a message is effectively protected from screenshot /
   158→   158→// forwarding. The effective flag is:
   159→   159→//   - the explicit `message.protected` value if it is a boolean
   160→   160→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   161→   161→//     the sender's settings here, so we just use the explicit value — the
   162→   162→//     backend always resolves and persists it on send, so the frontend
   163→   163→//     always sees the resolved boolean).
   164→   164→//
   165→   165→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   166→   166→// is true only when the current user is the recipient of a protected
   167→   167→// message and has NOT opted into "privacy always allow".
   168→   168→function useProtectionState(message: ChatMessage) {
   169→   169→  const me = useWaslStore((s) => s.user)
   170→   170→  const isOwner = !!me && message.senderId === me.id
   171→   171→  const alwaysAllow = !!me?.privacyAlwaysAllow
   172→   172→  const isProtected = message.protected === true
   173→   173→  const blocked = isProtected && !isOwner && !alwaysAllow
   174→   174→  return { isProtected, isOwner, alwaysAllow, blocked }
   175→   175→}

   177→   177→// Record a screenshot/copy/save attempt to the audit log via the API and
   178→   178→// show a toast. Idempotent — debounced per-message per-kind to avoid
   179→   179→// spamming the server when the user holds down PrintScreen.
   180→   180→const recentAttemptCache = new Map<string, number>()
   181→   181→function recordAttempt(messageId: string, kind: string) {
   182→   182→  const key = `${messageId}:${kind}`
   183→   183→  const now = Date.now()
   184→   184→  if (recentAttemptCache.has(key)) {
   185→   185→    const last = recentAttemptCache.get(key)!
   186→   186→    // Debounce: at most one record per 5 seconds per kind per message.
   187→   187→    if (now - last < 5000) return
   188→   188→  }
   189→   189→  recentAttemptCache.set(key, now)
   190→   190→  // Fire and forget
   191→   191→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   192→   192→    method: 'POST',
   193→   193→    headers: { 'Content-Type': 'application/json' },
   194→   194→    body: JSON.stringify({ kind }),
   195→   195→  }).catch(() => {})
   196→   196→}

   262→   262→  // ---- Keyboard capture-blocking (PrintScreen / Ctrl+C / Ctrl+S / Ctrl+P) ---
   263→   263→  // Mounted only when `blocked` is true. We attach a `keyup` listener to
   264→   264→  // `window` so we catch PrintScreen even when the bubble isn't focused, and
   265→   265→  // a `keydown` listener to the bubble element so we can preventDefault on
   266→   266→  // copy/save/print shortcuts when focus is inside the protected bubble.
   267→   267→  useEffect(() => {
   268→   268→    if (!blocked) return
   269→   269→    function onKeyUp(e: KeyboardEvent) {
   270→   270→      // PrintScreen key — most common screenshot trigger on Windows/Linux.
   271→   271→      if (e.key === 'PrintScreen') {
   272→   272→        toast.error(PROTECTED_WARNING, { duration: 4000 })
   273→   273→        recordAttempt(message.id, 'printscreen')
   274→   274→      }
   275→   275→    }
   276→   276→    function onKeyDown(e: KeyboardEvent) {
   277→   277→      // Ctrl/Cmd + C / S / P  → copy / save / print

   279→   279→      if (meta && (e.key === 'c' || e.key === 'C')) {
   280→   280→        const sel = window.getSelection?.()
   281→   281→        // Only block if the selection is inside our bubble.
   282→   282→        if (sel && bubbleRef.current && sel.anchorNode && bubbleRef.current.contains(sel.anchorNode)) {
   283→   283→          e.preventDefault()
   284→   284→          toast.error(PROTECTED_WARNING, { duration: 4000 })
   285→   285→          recordAttempt(message.id, 'copy')
   286→   286→        }

   289→   289→          e.preventDefault()
   290→   290→          toast.error(PROTECTED_WARNING, { duration: 4000 })
   291→   291→          recordAttempt(message.id, 'save')
   292→   292→        }
   293→   293→      }
   294→   294→    }
   295→   295→    window.addEventListener('keyup', onKeyUp)
   296→   296→    window.addEventListener('keydown', onKeyDown)
   297→   297→    return () => {
   298→   298→      window.removeEventListener('keyup', onKeyUp)
   299→   299→      window.removeEventListener('keydown', onKeyDown)

  1179→  1179→function groupReactions(reactions: Reaction[]) {
  1180→  1180→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1181→  1181→  for (const r of reactions) {
  1182→  1182→    const ex = map.get(r.emoji)
  1183→  1183→    if (ex) {
  1184→  1184→      ex.count++
  1185→  1185→      ex.users.push(r.userId)
  1186→  1186→    } else {
  1187→  1187→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1188→  1188→    }
  1189→  1189→  }
  1190→  1190→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
