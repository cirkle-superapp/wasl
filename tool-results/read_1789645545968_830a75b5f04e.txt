   534→// ==========================================================================
   535→// Official Service Providers — banks, government ministries, utilities.
   536→// These are manually authenticated and approved by Wasl admin staff.
   537→// Once approved, they can broadcast official announcements to all users
   538→// within a specific country.
   539→//
   540→// Registration flow:
   541→//   1. User applies → uploads official registration documents + passes
   542→//      liveness test (selfie video) + ID document
   543→//   2. Wasl admin manually reviews → approves or rejects
   544→//   3. Once approved, the provider gets a verified badge + broadcast rights
   545→//
   546→// Broadcast flow:
   547→//   1. Provider creates an announcement (text/image)
   548→//   2. System delivers to ALL users whose phone country code matches
   549→//   3. Users see official announcements in a dedicated "Official" tab
   550→// ==========================================================================
   551→
   552→model ServiceProvider {
   553→  id                String   @id @default(cuid())
   554→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   555→  ownerId           String   // the user who registered the provider
   556→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   557→  description       String   @default("")
   558→  avatarPath        String?
   559→  avatarColor       String?
   560→  // Official registration details
   561→  officialName      String?  // legal entity name
   562→  registrationNumber String? // government registration number
   563→  taxNumber         String?
   564→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   565→  // All users whose phone starts with this code receive broadcasts
   566→  countryCode       String   @default("+20")

   645→/// Scheduled message — send at a future time (Cirkle-inspired scheduled-messages)
   646→/// Supports recurring schedules: `repeat` can be 'none' (one-time), 'daily',
   647→/// 'weekly', or 'monthly'. When a recurring message is sent, `scheduledFor`
   648→/// is advanced to the next occurrence and `sent`/`sentAt` are reset so the
   649→/// background scheduler picks it up again at the next time.
   650→model ScheduledMessage {
   651→  id             String   @id @default(cuid())
   652→  conversationId String
   653→  senderId       String
   654→  content        String
   655→  type           String   @default("text")
   656→  scheduledFor   DateTime
   657→  sent           Boolean  @default(false)
   658→  sentAt         DateTime?
   659→  // Recurring schedule: 'none' = one-time, 'daily', 'weekly', 'monthly'.
   660→  // When non-'none', the scheduler advances `scheduledFor` to the next
   661→  // occurrence after sending instead of marking `sent=true`.
   662→  repeat         String   @default("none")
   663→  // Optional end date for recurring messages. After this date, the schedule
   664→  // stops (message is marked sent=true so the scheduler skips it).
   665→  repeatUntil    DateTime?
   666→  createdAt      DateTime @default(now())
   667→
   668→  @@index([scheduledFor])
   669→  @@index([conversationId])
   670→  @@index([senderId])
   671→}
