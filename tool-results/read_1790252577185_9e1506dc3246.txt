   684→   684→  // Toggles the bookmark state for a message. Optimistically updates the
   685→   685→  // `bookmarkedMessageIds` set in the store, then fires the API request.
   686→   686→  // The store always replaces the Set reference (instead of mutating) so
   687→   687→  // consumers subscribed to the set re-render correctly.
   688→   688→  const handleBookmark = useCallback(
   689→   689→    async (messageId: string) => {
   690→   690→      const isCurrentlyBookmarked = useWaslStore
   691→   691→        .getState()
   692→   692→        .bookmarkedMessageIds.has(messageId)
   693→   693→      // Optimistic update
   694→   694→      setBookmarked(messageId, !isCurrentlyBookmarked)
   695→   695→      try {
   696→   696→        if (isCurrentlyBookmarked) {
   697→   697→          // To DELETE a bookmark we need its row id. We don't store bookmark
   698→   698→          // ids in the store (only message ids), so re-fetch the list and
   699→   699→          // find the row matching this message.
   700→   700→          const res = await fetch('/api/bookmarks', { cache: 'no-store' })
   701→   701→          if (res.ok) {
   702→   702→            const data = await res.json()
   703→   703→            const found = (data.bookmarks || []).find(
   704→   704→              (b: { message: { id: string }; id: string }) =>
   705→   705→                b.message.id === messageId
   706→   706→            )
   707→   707→            if (found?.id) {
   708→   708→              await fetch(`/api/bookmarks/${found.id}`, { method: 'DELETE' })
   709→   709→              toast.success('Bookmark removed')
   710→   710→            } else {
   711→   711→              // Already removed server-side — reset store to be safe.
   712→   712→              setBookmarked(messageId, false)
   713→   713→            }
   714→   714→          }
   715→   715→        } else {
   716→   716→          const res = await fetch('/api/bookmarks', {
   717→   717→            method: 'POST',
   718→   718→            headers: { 'Content-Type': 'application/json' },
   719→   719→            body: JSON.stringify({ messageId }),
   720→   720→          })
   721→   721→          if (res.status === 409) {
   722→   722→            // Already bookmarked server-side — make sure store reflects that.
   723→   723→            setBookmarked(messageId, true)
   724→   724→          }
   725→   725→          if (res.ok || res.status === 409) {
   726→   726→            toast.success('Message bookmarked')
   727→   727→          } else {
   728→   728→            const data = await res.json().catch(() => null)
   729→   729→            toast.error(data?.error || 'Failed to bookmark')
   730→   730→            setBookmarked(messageId, false)
   731→   731→          }
   732→   732→        }
   733→   733→      } catch (e) {
   734→   734→        console.error(e)
   735→   735→        // Revert optimistic update on failure
   736→   736→        setBookmarked(messageId, isCurrentlyBookmarked)
   737→   737→        toast.error('Network error')
   738→   738→      }
   739→   739→    },
   740→   740→    [setBookmarked]
