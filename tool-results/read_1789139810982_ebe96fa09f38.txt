   231→// ─────────────────────────────────────────────────────────────────────────────
   232→// Mock QR matrix — deterministic 21x21 grid from a hash. Pure CSS so the
   233→// component is dependency-free. Renders three corner finder patterns so it
   234→// reads visually as a QR code.
   235→// ─────────────────────────────────────────────────────────────────────────────
   236→function HashQr({ hash, size = 132 }: { hash: string; size?: number }) {
   237→  const N = 21;
   238→  const cells: boolean[] = new Array(N * N).fill(false);
   239→  const hex = hash.replace(/[^0-9a-f]/gi, "");
   240→  for (let i = 0; i < cells.length; i++) {
   241→    const ch = hex.charCodeAt(i % Math.max(hex.length, 1));
   242→    cells[i] = (ch + i) % 2 === 0;
   243→  }
   244→  const stampFinder = (r0: number, c0: number) => {
   245→    for (let r = 0; r < 7; r++) {
   246→      for (let c = 0; c < 7; c++) {
   247→        const onBorder = r === 0 || r === 6 || c === 0 || c === 6;
   248→        const onInner = r >= 2 && r <= 4 && c >= 2 && c <= 4;
   249→        cells[(r0 + r) * N + (c0 + c)] = onBorder || onInner;
   250→      }
   251→    }
   252→  };
   253→  stampFinder(0, 0);
   254→  stampFinder(0, N - 7);
   255→  stampFinder(N - 7, 0);
   256→  return (
   257→    <div
   258→      className="grid bg-white p-2 rounded-md border border-border/60"
   259→      style={{
   260→        gridTemplateColumns: `repeat(${N}, 1fr)`,
   261→        gridTemplateRows: `repeat(${N}, 1fr)`,
   262→        width: size,
   263→        height: size,
   264→      }}
   265→      aria-label="Verification QR code"
   266→    >
   267→      {cells.map((on, i) => (
   268→        <div key={i} style={{ background: on ? "#1a1a14" : "white" }} />
   269→      ))}
   270→    </div>
   271→  );
   272→}

   274→// ─────────────────────────────────────────────────────────────────────────────
   275→// Countdown helper for jury cases
   276→// ─────────────────────────────────────────────────────────────────────────────
   277→function formatCountdown(expiresAt: string): string {
   278→  const ms = +new Date(expiresAt) - Date.now();
   279→  if (ms <= 0) return "Expired";
   280→  const h = Math.floor(ms / 3_600_000);
   281→  const m = Math.floor((ms % 3_600_000) / 60_000);
   282→  return h > 0 ? `${h}h ${m}m left` : `${m}m left`;
   283→}

   704→      toast.error("Commit the hash first");

   713→        body: JSON.stringify({
   714→          title: a.title,
   715→          description: a.description,

   720→          conditions: a.conditions,

   735→    } catch {
   736→      toast.error("Network error — try again");
   737→    } finally {
   738→      setVerifyingFor(null);
