   445→// ==========================================================================
   446→// Official Service Providers — banks, government ministries, utilities.
   447→// These are manually authenticated and approved by Wasl admin staff.
   448→// Once approved, they can broadcast official announcements to all users
   449→// within a specific country.
   450→//
   451→// Registration flow:
   452→//   1. User applies → uploads official registration documents + passes
   453→//      liveness test (selfie video) + ID document
   454→//   2. Wasl admin manually reviews → approves or rejects
   455→//   3. Once approved, the provider gets a verified badge + broadcast rights
   456→//
   457→// Broadcast flow:
   458→//   1. Provider creates an announcement (text/image)
   459→//   2. System delivers to ALL users whose phone country code matches
   460→//   3. Users see official announcements in a dedicated "Official" tab
   461→// ==========================================================================
   462→
   463→model ServiceProvider {
   464→  id                String   @id @default(cuid())
   465→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   466→  ownerId           String   // the user who registered the provider
   467→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   468→  description       String   @default("")
   469→  avatarPath        String?
   470→  avatarColor       String?
   471→  // Official registration details
   472→  officialName      String?  // legal entity name
   473→  registrationNumber String? // government registration number
   474→  taxNumber         String?
   475→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   476→  // All users whose phone starts with this code receive broadcasts
   477→  countryCode       String   @default("+20")
   478→  // Verification documents
   479→  registrationDocPath String?  // official registration document
   480→  idDocPath           String?  // ID of the authorized representative
   481→  livenessVideoPath  String?  // selfie video for liveness check
   482→  livenessVerified   Boolean  @default(false)
   483→  // Approval state — manually reviewed by Wasl admin
   484→  status            String   @default("pending") // pending | approved | rejected
   485→  rejectionReason   String?
   486→  verified          Boolean  @default(false)
   487→  verifiedAt        DateTime?
   488→  verifiedBy        String?  // admin user ID who approved
   489→  // Broadcast settings
   490→  canBroadcast      Boolean  @default(false)  // can send announcements
   491→  // Contact info
   492→  contactPhone      String?
   493→  contactEmail      String?
   494→  website           String?
   495→  createdAt         DateTime @default(now())
   496→  updatedAt         DateTime @updatedAt
   497→
   498→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   499→  broadcasts ServiceProviderMessage[]
   500→
   501→  @@index([ownerId])
   502→  @@index([countryCode])
   503→  @@index([status])
   504→  @@index([type])
   505→}
