   312→      }

   369→  // ---- Scroll to a specific message (by ID) ---------------------------------
   370→  // Used by the "jump to message" feature from the StarredMessagesDialog

   373→  useEffect(() => {
   374→    function onJumpToMessage(e: Event) {
   375→      const messageId = (e as CustomEvent<string>).detail
   376→      if (!messageId) return
   377→      // Prefer the scroll container scope, but fall back to a document-wide
   378→      // search if the ref isn't attached yet (e.g. during a re-render cycle)

   381→        (scrollRef.current?.querySelector(
   382→          `[data-message-id="${messageId}"]`
   383→        ) as HTMLElement | null) ||
   384→        (document.querySelector(
   385→          `[data-message-id="${messageId}"]`
   386→        ) as HTMLElement | null)
   387→      if (target) {
   388→        // Scroll the message into view (smooth, centered in the container)
   389→        target.scrollIntoView({ behavior: 'smooth', block: 'center' })
   390→        // Briefly flash the message to draw attention
   391→        target.classList.add('wasl-message-flash')
   392→        setTimeout(() => target.classList.remove('wasl-message-flash'), 2000)
   393→      }
   394→    }
   395→    // Remove a message from the local state after deletion (triggered by the
   396→    // DeleteMessageDialog after the API confirms the deletion).
   397→    function onMessageDeleted(e: Event) {

   410→    // it filtered out of this user's future message fetches.
   411→    function onMessageHiddenForMe(e: Event) {

   413→      if (!messageId || !activeConversationId) return
   414→      removeMessage(activeConversationId, messageId)

   426→  // ---- Load older messages ---------------------------------------------------

   442→        setHasMore(false)
   443→      } else {
   444→        prependMessages(activeConversationId, msgs)
   445→        msgs.forEach((m) => messageIdsRef.current.add(m.id))
   446→        setOldestLoaded(msgs[0].createdAt)
   447→        if (msgs.length < PAGE_SIZE) setHasMore(false)

   450→          if (scrollRef.current) {
   451→            const newHeight = scrollRef.current.scrollHeight
   452→            scrollRef.current.scrollTop =
   453→              newHeight - lastMessageHeightRef.current
   454→          }
   455→        })
   456→      }
   457→    } catch (e) {
   458→      console.error(e)
   459→    } finally {
   460→      setLoadingMore(false)
   461→    }
   462→  }

   485→      if (messageIdsRef.current.has(msg.id)) return

   489→      // Broadcast via socket

   491→      socket.emit('message:send', {

   502→      }

   505→      // ---- Demo companion bot: if this is a 1-on-1 chat with a demo user,
   506→      // simulate a typing indicator + contextual reply after a short delay.
   507→      if (
   508→        type === 'text' &&
   509→        conversation &&
   510→        !conversation.isGroup &&
   511→        otherUser &&
   512→        otherUser.phone.startsWith('+20100')
   513→      ) {
   514→        const conversationId = activeConversationId
   515→        const userId = user.id

   517→        // clients EXCEPT the sender, so we also set it locally to ensure the
   518→        // current user sees the typing indicator in the chat window and sidebar.
   519→        setTimeout(() => {
   520→          getSocket().emit('typing:start', {
   521→            conversationId,
   522→            userId: otherUser.userId,
   523→            name: otherUser.name,
   524→          })
   525→          useWaslStore.getState().setTyping(conversationId, otherUser.userId, true)
   526→        }, 600)
   527→        setBotReplying(true)
   528→        const delay = 1200 + Math.min(content.length * 30, 1800)
   529→        setTimeout(async () => {
   530→          getSocket().emit('typing:stop', {
   531→            conversationId,
   532→            userId: otherUser.userId,
   533→          })
   534→          useWaslStore.getState().setTyping(conversationId, otherUser.userId, false)
   535→          try {
   536→            const botRes = await fetch(
   537→              `/api/conversations/${conversationId}/bot-reply`,
   538→              { method: 'POST' }
   539→            )
   540→            if (!botRes.ok) return
   541→            const data = await botRes.json()
   542→            if (data.message && !messageIdsRef.current.has(data.message.id)) {
   543→              messageIdsRef.current.add(data.message.id)
   544→              addMessage(conversationId, data.message)
   545→              getSocket().emit('message:send', {
   546→                conversationId,
   547→                message: data.message,
   548→              })
   549→              if (conversation) {
   550→                upsertConversation({
   551→                  ...conversation,
   552→                  lastMessage: data.message,
   553→                  updatedAt: data.message.createdAt,
   554→                })
   555→              }
   556→            }
   557→          } catch (e) {
   558→            console.error(e)
   559→          } finally {
   560→            setBotReplying(false)
   561→          }
   562→        }, delay)
   563→      }
   564→    },
   565→    [activeConversationId, user?.id, replyTo, addMessage, setReplyTo, conversation, upsertConversation, otherUser]
   566→  )
