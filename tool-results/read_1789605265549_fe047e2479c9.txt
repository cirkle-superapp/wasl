   199→  async function toggleMute() {
   200→    const next = !muted

   202→    try {
   203→      const res = await fetch(
   204→        `/api/conversations/${activeConversationId}/mute`,
   205→        {
   206→          method: 'POST',
   207→          headers: { 'Content-Type': 'application/json' },
   208→          body: JSON.stringify({ muted: next }),
   209→        }
   210→      )
   211→      if (!res.ok) {
   212→        setMuted(!next) // rollback
   213→        toast.error('Failed to update mute setting')
   214→        return
   215→      }
   216→      toast.success(next ? 'Notifications muted' : 'Notifications unmuted')
   217→    } catch {
   218→      setMuted(!next) // rollback
   219→      toast.error('Network error')
   220→    }

   240→  // Remove a member from the group (admin only).
   241→  async function handleRemoveMember(userId: string, name: string) {
   242→    if (!activeConversationId) return
   243→    if (
   244→      !confirm(
   245→        `Remove ${name} from this group? They will no longer be able to see or send messages here.`
   246→      )
   247→    )
   248→      return
   249→    setRemovingUserId(userId)
   250→    try {
   251→      const res = await fetch(
   252→        `/api/conversations/${activeConversationId}/members?userId=${encodeURIComponent(userId)}`,
   253→        { method: 'DELETE' }
   254→      )
   255→      if (!res.ok) {
   256→        const data = await res.json().catch(() => null)
   257→        toast.error(data?.error || 'Failed to remove member')
   258→        return
   259→      }
   260→      // Optimistic local update — drop the removed member from the list
   261→      setLocalParticipants((prev) => {
   262→        const base = prev ?? conversation?.participants ?? []
   263→        return base.filter((p) => p.userId !== userId)
   264→      })
   265→      toast.success(`${name} removed`)
   266→      // Refetch so the sidebar member count + last-message preview refresh.
   267→      void refreshConversation()
   268→    } catch {
   269→      toast.error('Network error')
   270→    } finally {
   271→      setRemovingUserId(null)
   272→    }
   273→  }

   822→
   823→type SearchUser = {
   824→  id: string
   825→  username: string
   826→  name: string
   827→  phone: string | null
   828→  avatar: string | null
   829→  avatarColor: string | null
   830→  about: string
   831→  online: boolean
   832→  lastSeen: string
   833→  verified: boolean

   835→
   836→function AddMemberDialog({
   837→  open,
   838→  onOpenChange,
   839→  conversationId,
   840→  currentMemberIds,
   841→  onMemberAdded,
   842→}: {
   843→  open: boolean
   844→  onOpenChange: (v: boolean) => void
   845→  conversationId: string
   846→  currentMemberIds: string[]
   847→  onMemberAdded: (u: SearchUser) => void
   848→}) {
   849→  const [query, setQuery] = useState('')
   850→  const [results, setResults] = useState<SearchUser[]>([])
   851→  const [loading, setLoading] = useState(false)
   852→  const [addingId, setAddingId] = useState<string | null>(null)
   853→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   854→  const memberSet = useRef<Set<string>>(new Set(currentMemberIds))
   855→
   856→  // Keep the current-member Set in sync whenever the prop changes (e.g. after
   857→  // an optimistic add/remove in the parent).
   858→  useEffect(() => {
   859→    memberSet.current = new Set(currentMemberIds)
   860→  }, [currentMemberIds])
   861→
   862→  // Reset search state whenever the dialog closes so it's clean on re-open.
   863→  useEffect(() => {
   864→    if (!open) {
   865→      setQuery('')
   866→      setResults([])
   867→      setAddingId(null)
   868→      if (debounceRef.current) {
   869→        clearTimeout(debounceRef.current)
   870→        debounceRef.current = null
   871→      }
   872→    }
   873→  }, [open])
   874→
   875→  // Debounced search — fires 300ms after the user stops typing.
   876→  useEffect(() => {
   877→    if (!open) return
   878→    const q = query.trim()
   879→    if (!q) {
   880→      setResults([])
   881→      setLoading(false)
   882→      return
   883→    }
   884→    setLoading(true)
   885→    if (debounceRef.current) clearTimeout(debounceRef.current)
   886→    debounceRef.current = setTimeout(async () => {
   887→      try {
   888→        const res = await fetch(
   889→          `/api/users/search?q=${encodeURIComponent(q)}`,
   890→          { cache: 'no-store' }
   891→        )
   892→        if (!res.ok) {
   893→          setResults([])
   894→          return
   895→        }
   896→        const data = await res.json()
   897→        setResults((data.users as SearchUser[]) || [])
   898→      } catch {
   899→        setResults([])
   900→      } finally {
   901→        setLoading(false)
   902→      }
   903→    }, 300)
   904→    return () => {
   905→      if (debounceRef.current) {
   906→        clearTimeout(debounceRef.current)
   907→        debounceRef.current = null
   908→      }
   909→    }
   910→  }, [query, open])
   911→
   912→  async function handleAdd(user: SearchUser) {
   913→    setAddingId(user.id)
   914→    try {
   915→      const res = await fetch(
   916→        `/api/conversations/${conversationId}/members`,
   917→        {
   918→          method: 'POST',
   919→          headers: { 'Content-Type': 'application/json' },
   920→          body: JSON.stringify({ userId: user.id }),
   921→        }
   922→      )
   923→      if (!res.ok) {
   924→        const data = await res.json().catch(() => null)
   925→        toast.error(data?.error || 'Failed to add member')
   926→        return
   927→      }
   928→      toast.success(`${user.name} added to the group`)
   929→      onMemberAdded(user)
   930→      // Remove from the visible search results so the user can't add twice.
   931→      setResults((prev) => prev.filter((u) => u.id !== user.id))
   932→      // Close after a successful add.
   933→      onOpenChange(false)
   934→    } catch {
   935→      toast.error('Network error')
   936→    } finally {
   937→      setAddingId(null)
   938→    }
   939→  }
   940→
   941→  return (
   942→    <Dialog open={open} onOpenChange={onOpenChange}>
   943→      <DialogContent className="max-w-md max-h-[80vh] flex flex-col p-0 gap-0">
   944→        <DialogHeader className="px-4 pt-4 pb-2">
   945→          <DialogTitle>Add member</DialogTitle>
   946→          <DialogDescription>
   947→            Search for a Wasl user by name, username, or phone number. They
   948→            will be added to this group as a regular member.
   949→          </DialogDescription>
   950→        </DialogHeader>
   951→        <div className="px-4 pb-2">
   952→          <div className="relative">
   953→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground pointer-events-none" />
   954→            <Input
   955→              autoFocus
   956→              value={query}
   957→              onChange={(e) => setQuery(e.target.value)}
   958→              placeholder="Search name, username, or phone…"
   959→              className="pl-9"
   960→            />
   961→            {loading && (
   962→              <Loader2 className="absolute right-2.5 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />
   963→            )}
   964→          </div>
   965→        </div>
   966→        <div className="flex-1 overflow-y-auto wasl-scroll px-2 pb-2">
   967→          {!loading && query.trim() === '' ? (
   968→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
   969→              Start typing to search for someone to add.
   970→            </div>
   971→          ) : !loading && results.length === 0 ? (
   972→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
   973→              No users found for &ldquo;{query}&rdquo;.
   974→            </div>
   975→          ) : (
   976→            <div className="space-y-0.5">
   977→              {results.map((u) => {
   978→                const alreadyMember = memberSet.current.has(u.id)
   979→                return (
   980→                  <button
   981→                    key={u.id}
   982→                    type="button"
   983→                    disabled={alreadyMember || addingId === u.id}
   984→                    onClick={() => handleAdd(u)}
   985→                    className={cn(
   986→                      'w-full flex items-center gap-3 p-2 rounded-lg text-left transition-colors',
   987→                      alreadyMember
   988→                        ? 'opacity-50 cursor-not-allowed'
   989→                        : 'hover:bg-muted/60 cursor-pointer'
   990→                    )}
   991→                  >
   992→                    <WaslAvatar
   993→                      name={u.name}
   994→                      src={u.avatar}
   995→                      color={u.avatarColor}
   996→                      size={36}
   997→                      online={u.online}
   998→                      showStatus
   999→                    />
  1000→                    <div className="flex-1 min-w-0">
  1001→                      <div className="text-sm font-medium truncate">{u.name}</div>
  1002→                      <div className="text-xs text-muted-foreground truncate">
  1003→                        @{u.username}
  1004→                        {u.phone ? ` · ${u.phone}` : ''}
  1005→                      </div>
  1006→                    </div>
  1007→                    {alreadyMember ? (
  1008→                      <span className="text-[10px] text-muted-foreground shrink-0">
  1009→                        Already in group
  1010→                      </span>
  1011→                    ) : addingId === u.id ? (
  1012→                      <Loader2 className="w-4 h-4 animate-spin text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1013→                    ) : (
  1014→                      <UserPlus className="w-4 h-4 text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1015→                    )}
  1016→                  </button>
  1017→                )
  1018→              })}
  1019→            </div>
  1020→          )}
  1021→        </div>
  1022→      </DialogContent>
  1023→    </Dialog>
  1024→  )

  1032→function RenameGroupDialog({

  1040→  onOpenChange: (v: boolean) => void

  1104→            {trimmed.length}/100 characters
