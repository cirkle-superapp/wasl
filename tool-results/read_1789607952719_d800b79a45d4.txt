    46→// ---- Search highlight helper ---------------------------------------------
    47→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    48→// match in a <mark> with a visible yellow background so the user can see why
    49→// the message was returned by the chat-search dialog.
    50→//
    51→// Supports multi-word queries: if the query contains spaces, each word is
    52→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    53→// and "Wasl" separately). Each word is regex-escaped so special characters
    54→// (`(`, `.`, `*`, …) are matched literally.
    55→//
    56→// Returns the original `text` (a single string node) when either the text or
    57→// query is empty, so plain messages with no active query render with zero
    58→// overhead and no behavioural change vs. before.
    59→function highlightText(text: string, query: string): React.ReactNode {
    60→  if (!text) return text
    61→  const q = query.trim()
    62→  if (!q) return text
    63→
    64→  // Split the query into individual words for multi-word highlighting.
    65→  // Filter out empty strings from multiple consecutive spaces.
    66→  const words = q.split(/\s+/).filter(Boolean)
    67→  if (words.length === 0) return text
    68→
    69→  // Build a single regex that matches ANY of the words (OR).
    70→  // Each word is regex-escaped so special characters are literal.
    71→  const escapedWords = words.map((w) =>
    72→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    73→  )
    74→  const pattern = escapedWords.join('|')
    75→  let regex: RegExp
    76→  try {
    77→    regex = new RegExp(`(${pattern})`, 'gi')
    78→  } catch {
    79→    // Malformed query — bail to plain text rather than crashing the bubble.
    80→    return text
    81→  }
    82→  // String.split with a capturing group keeps the matched delimiters in the
    83→  // result array, so odd indices are the query matches (highlight them) and
    84→  // even indices are the surrounding plain text.
    85→  const parts = text.split(regex)
    86→  if (parts.length === 1) {
    87→    // No matches — render as a single plain string node.
    88→    return text
    89→  }
    90→  return parts.map((part, i) =>
    91→    i % 2 === 1 ? (
    92→      <mark
    93→        key={i}
    94→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    95→      >
    96→        {part}
    97→      </mark>
    98→    ) : (
    99→      // Render empty strings as `null` so React doesn't complain about
   100→      // adjacent text separators. Non-empty even-indexed parts become plain
   101→      // string nodes (no wrapping element needed).
   102→      part || null
   103→    )
   104→  )
   105→}

   107→// ---- File-attachment content parser --------------------------------------
   108→// PDF / document / audio messages store their payload as a small JSON blob
   109→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   110→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   111→// resilient when the JSON shape isn't present.
   112→export function parseFileContent(
   113→  content: string
   114→): { url: string; name?: string; size?: number } | null {
   115→  if (!content) return null
   116→  try {
   117→    const parsed = JSON.parse(content)
   118→    if (
   119→      parsed &&
   120→      typeof parsed === 'object' &&
   121→      typeof parsed.url === 'string'
   122→    ) {
   123→      return {
   124→        url: parsed.url,
   125→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   126→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   127→      }
   128→    }
   129→  } catch {
   130→    // Not JSON — fall through to plain-URL detection.
   131→  }
   132→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   133→  if (content.startsWith('http') || content.startsWith('/')) {
   134→    return { url: content }
   135→  }
   136→  return null
   137→}

   139→export function formatFileSize(bytes: number): string {
   140→  if (bytes < 1024) return `${bytes} B`
   141→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   142→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   143→}

   145→// Derive a filename from a URL when the parsed payload has no `name`.
   146→function deriveFilenameFromUrl(url: string, fallback: string): string {
   147→  try {
   148→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   149→    const last = u.pathname.split('/').filter(Boolean).pop()
   150→    return last ? decodeURIComponent(last) : fallback
   151→  } catch {
   152→    return fallback
   153→  }

   156→// ---- Protection helpers -------------------------------------------------
   157→// Resolves whether a message is effectively protected from screenshot /
   158→// forwarding. The effective flag is:
   159→//   - the explicit `message.protected` value if it is a boolean
   160→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   161→//     the sender's settings here, so we just use the explicit value — the
   162→//     backend always resolves and persists it on send, so the frontend
   163→//     always sees the resolved boolean).
   164→//
   165→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   166→// is true only when the current user is the recipient of a protected
   167→// message and has NOT opted into "privacy always allow".
   168→function useProtectionState(message: ChatMessage) {
   169→  const me = useWaslStore((s) => s.user)
   170→  const isOwner = !!me && message.senderId === me.id
   171→  const alwaysAllow = !!me?.privacyAlwaysAllow
   172→  const isProtected = message.protected === true
   173→  const blocked = isProtected && !isOwner && !alwaysAllow
   174→  return { isProtected, isOwner, alwaysAllow, blocked }
   175→}

   177→// Record a screenshot/copy/save attempt to the audit log via the API and
   178→// show a toast. Idempotent — debounced per-message per-kind to avoid
   179→// spamming the server when the user holds down PrintScreen.
   180→const recentAttemptCache = new Map<string, number>()
   181→function recordAttempt(messageId: string, kind: string) {
   182→  const key = `${messageId}:${kind}`
   183→  const now = Date.now()
   184→  if (recentAttemptCache.has(key)) {
   185→    const last = recentAttemptCache.get(key)!
   186→    // Debounce: at most one record per 5 seconds per kind per message.
   187→    if (now - last < 5000) return
   188→  }
   189→  recentAttemptCache.set(key, now)
   190→  // Fire and forget
   191→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   192→    method: 'POST',
   193→    headers: { 'Content-Type': 'application/json' },
   194→    body: JSON.stringify({ kind }),
   195→  }).catch(() => {})
   196→}

   917→// ---- File-attachment card (PDF / document) -------------------------------
   918→// Renders a compact "file card" inside the bubble: a coloured FileText icon
   919→// (red for PDF, teal/green for documents), the original filename + human-
   920→// readable size, and a Download button that opens the URL in a new tab.

   922→// `blocked` is set when the message is protected AND the current user is the

   942→    parsed?.name ||
   943→    (parsed?.url ? deriveFilenameFromUrl(parsed.url, fallbackName) : fallbackName)
   944→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : isPdf ? 'PDF' : 'Document'

   947→  // respects light/dark mode and the Cirkle gold theme). No indigo/blue.

   985→            href={parsed?.url}

   987→            rel="noopener noreferrer"
   988→            download={name}

   991→              'shrink-0 px-2.5 py-1.5 rounded-md text-xs font-medium transition-colors',
   992→              'bg-[var(--wasl-green)]/10 text-[var(--wasl-green)] hover:bg-[var(--wasl-green)]/20'

  1005→      </div>
  1006→    </div>

  1179→function groupReactions(reactions: Reaction[]) {
  1180→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1181→  for (const r of reactions) {
  1182→    const ex = map.get(r.emoji)
  1183→    if (ex) {
  1184→      ex.count++
  1185→      ex.users.push(r.userId)
  1186→    } else {
  1187→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1188→    }
  1189→  }
  1190→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
  1191→}
