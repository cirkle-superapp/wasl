   414→  // Upload a new group avatar (admin only). Posts the file to /api/upload,
   415→  // then PATCHes the conversation with the returned URL. Reuses the
   416→  // 5 MB cap from the upload API for the client-side guard so we fail fast
   417→  // before the network round-trip.
   418→  async function handleAvatarFileChange(
   419→    e: React.ChangeEvent<HTMLInputElement>
   420→  ) {
   421→    const file = e.target.files?.[0]
   422→    // Always clear the input value so picking the same file twice in a row
   423→    // still fires onChange.
   424→    e.target.value = ''
   425→    if (!file) return
   426→    if (!activeConversationId) return
   427→    if (!conversation?.isGroup || !isAdmin) return
   428→
   429→    if (file.size > 5 * 1024 * 1024) {
   430→      toast.error('Image too large (max 5MB)')
   431→      return
   432→    }
   433→    if (!/^image\//i.test(file.type)) {
   434→      toast.error('Only image files are allowed')
   435→      return
   436→    }
   437→
   438→    setAvatarUploading(true)
   439→    try {
   440→      // Step 1: upload the image bytes to /api/upload.
   441→      const form = new FormData()
   442→      form.append('file', file)
   443→      const upRes = await fetch('/api/upload?type=group-avatar', {
   444→        method: 'POST',
   445→        body: form,
   446→      })
   447→      if (!upRes.ok) {
   448→        const upData = await upRes.json().catch(() => null)
   449→        toast.error(upData?.error || 'Failed to upload image')
   450→        return
   451→      }
   452→      const upJson = (await upRes.json()) as { url?: string }
   453→      const url = upJson.url
   454→      if (!url) {
   455→        toast.error('Upload response is missing the image URL')
   456→        return
   457→      }
   458→
   459→      // Step 2: PATCH the conversation with the returned URL.
   460→      const patchRes = await fetch(
   461→        `/api/conversations/${activeConversationId}`,
   462→        {
   463→          method: 'PATCH',
   464→          headers: { 'Content-Type': 'application/json' },
   465→          body: JSON.stringify({ avatar: url }),
   466→        }
   467→      )
   468→      if (!patchRes.ok) {
   469→        const patchData = await patchRes.json().catch(() => null)
   470→        toast.error(patchData?.error || 'Failed to set group photo')
   471→        return
   472→      }
   473→      toast.success('Group avatar updated')
   474→      handleAvatarChanged(url)
   475→    } catch {
   476→      toast.error('Network error')
   477→    } finally {
   478→      setAvatarUploading(false)
   479→    }
   480→  }

  1145→// AddMemberDialog — search users and add them to the group (admin only).
  1146→// Excludes existing members from the search results. Calls
  1147→// POST /api/conversations/:id/members with { userId } when a search result
  1148→// is clicked.
  1149→// ---------------------------------------------------------------------------
  1150→
  1151→type SearchUser = {
  1152→  id: string
  1153→  username: string
  1154→  name: string
  1155→  phone: string | null
  1156→  avatar: string | null
  1157→  avatarColor: string | null
  1158→  about: string
  1159→  online: boolean
  1160→  lastSeen: string
  1161→  verified: boolean
  1162→}
  1163→
  1164→function AddMemberDialog({
  1165→  open,
  1166→  onOpenChange,
  1167→  conversationId,
  1168→  currentMemberIds,
  1169→  onMemberAdded,
  1170→}: {
  1171→  open: boolean
  1172→  onOpenChange: (v: boolean) => void
  1173→  conversationId: string
  1174→  currentMemberIds: string[]
  1175→  onMemberAdded: (u: SearchUser) => void
  1176→}) {
  1177→  const [query, setQuery] = useState('')
  1178→  const [results, setResults] = useState<SearchUser[]>([])
  1179→  const [loading, setLoading] = useState(false)
  1180→  const [addingId, setAddingId] = useState<string | null>(null)
  1181→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
  1182→  const memberSet = useRef<Set<string>>(new Set(currentMemberIds))
  1183→
  1184→  // Keep the current-member Set in sync whenever the prop changes (e.g. after
  1185→  // an optimistic add/remove in the parent).
  1186→  useEffect(() => {
  1187→    memberSet.current = new Set(currentMemberIds)
  1188→  }, [currentMemberIds])
  1189→
  1190→  // Reset search state whenever the dialog closes so it's clean on re-open.
  1191→  useEffect(() => {
  1192→    if (!open) {
  1193→      setQuery('')
  1194→      setResults([])
  1195→      setAddingId(null)
  1196→      if (debounceRef.current) {
  1197→        clearTimeout(debounceRef.current)
  1198→        debounceRef.current = null
  1199→      }
  1200→    }
  1201→  }, [open])
  1202→
  1203→  // Debounced search — fires 300ms after the user stops typing.
  1204→  useEffect(() => {
  1205→    if (!open) return
  1206→    const q = query.trim()
  1207→    if (!q) {
  1208→      setResults([])
  1209→      setLoading(false)
  1210→      return
  1211→    }
  1212→    setLoading(true)
  1213→    if (debounceRef.current) clearTimeout(debounceRef.current)
  1214→    debounceRef.current = setTimeout(async () => {
  1215→      try {
  1216→        const res = await fetch(
  1217→          `/api/users/search?q=${encodeURIComponent(q)}`,
  1218→          { cache: 'no-store' }
  1219→        )
  1220→        if (!res.ok) {
  1221→          setResults([])
  1222→          return
  1223→        }
  1224→        const data = await res.json()
  1225→        setResults((data.users as SearchUser[]) || [])
  1226→      } catch {
  1227→        setResults([])
  1228→      } finally {
  1229→        setLoading(false)
  1230→      }
  1231→    }, 300)
  1232→    return () => {
  1233→      if (debounceRef.current) {
  1234→        clearTimeout(debounceRef.current)
  1235→        debounceRef.current = null
  1236→      }
  1237→    }
  1238→  }, [query, open])
  1239→
  1240→  async function handleAdd(user: SearchUser) {
  1241→    setAddingId(user.id)
  1242→    try {
  1243→      const res = await fetch(
  1244→        `/api/conversations/${conversationId}/members`,
  1245→        {
  1246→          method: 'POST',
  1247→          headers: { 'Content-Type': 'application/json' },
  1248→          body: JSON.stringify({ userId: user.id }),
  1249→        }
  1250→      )
  1251→      if (!res.ok) {
  1252→        const data = await res.json().catch(() => null)
  1253→        toast.error(data?.error || 'Failed to add member')
  1254→        return
  1255→      }
  1256→      toast.success(`${user.name} added to the group`)
  1257→      onMemberAdded(user)
  1258→      // Remove from the visible search results so the user can't add twice.
  1259→      setResults((prev) => prev.filter((u) => u.id !== user.id))
  1260→      // Close after a successful add.
  1261→      onOpenChange(false)
  1262→    } catch {
  1263→      toast.error('Network error')
  1264→    } finally {
  1265→      setAddingId(null)
  1266→    }
  1267→  }
  1268→
  1269→  return (
  1270→    <Dialog open={open} onOpenChange={onOpenChange}>
  1271→      <DialogContent className="max-w-md max-h-[80vh] flex flex-col p-0 gap-0">
  1272→        <DialogHeader className="px-4 pt-4 pb-2">
  1273→          <DialogTitle>Add member</DialogTitle>
  1274→          <DialogDescription>
  1275→            Search for a Wasl user by name, username, or phone number. They
  1276→            will be added to this group as a regular member.
  1277→          </DialogDescription>
  1278→        </DialogHeader>
  1279→        <div className="px-4 pb-2">
  1280→          <div className="relative">
  1281→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground pointer-events-none" />
  1282→            <Input
  1283→              autoFocus
  1284→              value={query}
  1285→              onChange={(e) => setQuery(e.target.value)}
  1286→              placeholder="Search name, username, or phone…"
  1287→              className="pl-9"
  1288→            />
  1289→            {loading && (
  1290→              <Loader2 className="absolute right-2.5 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />
  1291→            )}
  1292→          </div>
  1293→        </div>
  1294→        <div className="flex-1 overflow-y-auto wasl-scroll px-2 pb-2">
  1295→          {!loading && query.trim() === '' ? (
  1296→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
  1297→              Start typing to search for someone to add.
  1298→            </div>
  1299→          ) : !loading && results.length === 0 ? (
  1300→            <div className="px-2 py-6 text-center text-sm text-muted-foreground">
  1301→              No users found for &ldquo;{query}&rdquo;.
  1302→            </div>
  1303→          ) : (
  1304→            <div className="space-y-0.5">
  1305→              {results.map((u) => {
  1306→                const alreadyMember = memberSet.current.has(u.id)
  1307→                return (
  1308→                  <button
  1309→                    key={u.id}
  1310→                    type="button"
  1311→                    disabled={alreadyMember || addingId === u.id}
  1312→                    onClick={() => handleAdd(u)}
  1313→                    className={cn(
  1314→                      'w-full flex items-center gap-3 p-2 rounded-lg text-left transition-colors',
  1315→                      alreadyMember
  1316→                        ? 'opacity-50 cursor-not-allowed'
  1317→                        : 'hover:bg-muted/60 cursor-pointer'
  1318→                    )}
  1319→                  >
  1320→                    <WaslAvatar
  1321→                      name={u.name}
  1322→                      src={u.avatar}
  1323→                      color={u.avatarColor}
  1324→                      size={36}
  1325→                      online={u.online}
  1326→                      showStatus
  1327→                    />
  1328→                    <div className="flex-1 min-w-0">
  1329→                      <div className="text-sm font-medium truncate">{u.name}</div>
  1330→                      <div className="text-xs text-muted-foreground truncate">
  1331→                        @{u.username}
  1332→                        {u.phone ? ` · ${u.phone}` : ''}
  1333→                      </div>
  1334→                    </div>
  1335→                    {alreadyMember ? (
  1336→                      <span className="text-[10px] text-muted-foreground shrink-0">
  1337→                        Already in group
  1338→                      </span>
  1339→                    ) : addingId === u.id ? (
  1340→                      <Loader2 className="w-4 h-4 animate-spin text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1341→                    ) : (
  1342→                      <UserPlus className="w-4 h-4 text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" />
  1343→                    )}
  1344→                  </button>
  1345→                )
  1346→              })}
  1347→            </div>
  1348→          )}
  1349→        </div>
  1350→      </DialogContent>
  1351→    </Dialog>
  1352→  )
  1353→}

  1356→// RenameGroupDialog — inline form to rename the group (admin only).

  1360→function RenameGroupDialog({
  1361→  open,
  1362→  onOpenChange,
  1363→  conversationId,
  1364→  currentName,
  1365→  onRenamed,
  1366→}: {
  1367→  open: boolean
  1368→  onOpenChange: (v: boolean) => void
  1369→  conversationId: string
  1370→  currentName: string
  1371→  onRenamed: (newName: string) => void
  1372→}) {

  1375→

  1459→// EditDescriptionDialog — textarea form to set or clear the group description

  1461→// An empty (whitespace-only) textarea is treated as "clear description".
  1462→// ---------------------------------------------------------------------------
  1463→
  1464→function EditDescriptionDialog({
  1465→  open,
  1466→  onOpenChange,
  1467→  conversationId,
  1468→  currentDescription,
  1469→  onSaved,
  1470→}: {
  1471→  open: boolean
  1472→  onOpenChange: (v: boolean) => void
  1473→  conversationId: string
  1474→  currentDescription: string | null
  1475→  onSaved: (newDescription: string | null) => void
  1476→}) {
  1477→  const [value, setValue] = useState(currentDescription ?? '')
  1478→  const [saving, setSaving] = useState(false)
  1479→
  1480→  // Re-seed the textarea whenever the dialog opens so it always reflects
  1481→  // the canonical description (not a stale draft from a previous open).
  1482→  useEffect(() => {
  1483→    if (open) setValue(currentDescription ?? '')
  1484→  }, [open, currentDescription])
  1485→
  1486→  const trimmed = value.trim()
  1487→  // Valid as long as it's within the 500-char limit. Empty is allowed (clears).
  1488→  const valid = value.length <= 500
  1489→  // Disable Save if nothing changed (same content after trim) or invalid.
  1490→  const previousTrimmed = (currentDescription ?? '').trim()
  1491→  const changed = trimmed !== previousTrimmed
  1492→
  1493→  async function handleSave() {
  1494→    if (!valid || !changed) return
  1495→    setSaving(true)
  1496→    try {
  1497→      const res = await fetch(`/api/conversations/${conversationId}`, {
  1498→        method: 'PATCH',
  1499→        headers: { 'Content-Type': 'application/json' },
  1500→        body: JSON.stringify({ description: value }),
  1501→      })
  1502→      if (!res.ok) {
  1503→        const data = await res.json().catch(() => null)
  1504→        toast.error(data?.error || 'Failed to update description')
  1505→        return
  1506→      }
  1507→      const next = trimmed.length > 0 ? trimmed : null
  1508→      toast.success(
  1509→        next ? 'Description updated' : 'Description deleted'
  1510→      )
  1511→      onSaved(next)
  1512→      onOpenChange(false)
  1513→    } catch {
  1514→      toast.error('Network error')
  1515→    } finally {
  1516→      setSaving(false)
  1517→    }
  1518→  }
  1519→
  1520→  return (
  1521→    <Dialog open={open} onOpenChange={onOpenChange}>
  1522→      <DialogContent className="max-w-md">
  1523→        <DialogHeader>
  1524→          <DialogTitle>
  1525→            {currentDescription ? 'Edit description' : 'Add description'}
  1526→          </DialogTitle>
  1527→          <DialogDescription>
  1528→            Group descriptions are visible to all members. Line breaks are
  1529→            preserved. Leave empty to clear the description.
  1530→          </DialogDescription>
  1531→        </DialogHeader>
  1532→        <div className="space-y-2">
  1533→          <Label htmlFor="wasl-group-description-input">Description</Label>
  1534→          <Textarea
  1535→            id="wasl-group-description-input"
  1536→            autoFocus
  1537→            value={value}
  1538→            onChange={(e) => setValue(e.target.value)}
  1539→            maxLength={500}
  1540→            placeholder="What is this group about?"
  1541→            className="min-h-24 resize-y"
  1542→          />
  1543→          <div className="flex items-center justify-between text-[11px] text-muted-foreground">
  1544→            <span>
  1545→              {trimmed.length === 0 && value.length > 0
  1546→                ? 'Only whitespace — will clear the description'
  1547→                : 'Line breaks are preserved.'}
  1548→            </span>
  1549→            <span className={cn(value.length > 500 && 'text-destructive')}>
  1550→              {value.length}/500
  1551→            </span>
  1552→          </div>
  1553→        </div>
  1554→        <DialogFooter>
  1555→          <Button
  1556→            variant="ghost"
  1557→            onClick={() => onOpenChange(false)}
  1558→            disabled={saving}
  1559→          >
  1560→            Cancel
  1561→          </Button>
  1562→          <Button
  1563→            onClick={handleSave}
  1564→            disabled={!valid || !changed || saving}
  1565→          >
  1566→            {saving ? (
  1567→              <>
  1568→                <Loader2 className="w-4 h-4 mr-2 animate-spin" /> Saving…
  1569→              </>
  1570→            ) : (
  1571→              'Save'
  1572→            )}
  1573→          </Button>
  1574→        </DialogFooter>
  1575→      </DialogContent>
  1576→    </Dialog>
  1577→  )
  1578→}
