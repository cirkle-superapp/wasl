    46→// ---- Search highlight helper ---------------------------------------------
    47→// Splits `text` on case-insensitive occurrences of `query` and wraps each
    48→// match in a <mark> with a visible yellow background so the user can see why
    49→// the message was returned by the chat-search dialog.
    50→//
    51→// Supports multi-word queries: if the query contains spaces, each word is
    52→// highlighted independently (e.g. "welcome wasl" highlights both "Welcome"
    53→// and "Wasl" separately). Each word is regex-escaped so special characters
    54→// (`(`, `.`, `*`, …) are matched literally.
    55→//
    56→// Returns the original `text` (a single string node) when either the text or
    57→// query is empty, so plain messages with no active query render with zero
    58→// overhead and no behavioural change vs. before.
    59→function highlightText(text: string, query: string): React.ReactNode {
    60→  if (!text) return text
    61→  const q = query.trim()
    62→  if (!q) return text
    63→
    64→  // Split the query into individual words for multi-word highlighting.
    65→  // Filter out empty strings from multiple consecutive spaces.
    66→  const words = q.split(/\s+/).filter(Boolean)
    67→  if (words.length === 0) return text
    68→
    69→  // Build a single regex that matches ANY of the words (OR).
    70→  // Each word is regex-escaped so special characters are literal.
    71→  const escapedWords = words.map((w) =>
    72→    w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    73→  )
    74→  const pattern = escapedWords.join('|')
    75→  let regex: RegExp
    76→  try {
    77→    regex = new RegExp(`(${pattern})`, 'gi')
    78→  } catch {
    79→    // Malformed query — bail to plain text rather than crashing the bubble.
    80→    return text
    81→  }
    82→  // String.split with a capturing group keeps the matched delimiters in the
    83→  // result array, so odd indices are the query matches (highlight them) and
    84→  // even indices are the surrounding plain text.
    85→  const parts = text.split(regex)
    86→  if (parts.length === 1) {
    87→    // No matches — render as a single plain string node.
    88→    return text
    89→  }
    90→  return parts.map((part, i) =>
    91→    i % 2 === 1 ? (
    92→      <mark
    93→        key={i}
    94→        className="bg-yellow-200 dark:bg-yellow-900/70 text-foreground rounded px-0.5"
    95→      >
    96→        {part}
    97→      </mark>
    98→    ) : (
    99→      // Render empty strings as `null` so React doesn't complain about
   100→      // adjacent text separators. Non-empty even-indexed parts become plain
   101→      // string nodes (no wrapping element needed).
   102→      part || null
   103→    )
   104→  )
   105→}

   107→// ---- File-attachment content parser --------------------------------------
   108→// PDF / document / audio messages store their payload as a small JSON blob
   109→// in the message `content` field: `{ url, name, size }`. This parser tolerates
   110→// a plain URL string too (legacy / forwarded payloads) so the renderer is
   111→// resilient when the JSON shape isn't present.
   112→export function parseFileContent(
   113→  content: string
   114→): { url: string; name?: string; size?: number } | null {
   115→  if (!content) return null
   116→  try {
   117→    const parsed = JSON.parse(content)
   118→    if (
   119→      parsed &&
   120→      typeof parsed === 'object' &&
   121→      typeof parsed.url === 'string'
   122→    ) {
   123→      return {
   124→        url: parsed.url,
   125→        name: typeof parsed.name === 'string' ? parsed.name : undefined,
   126→        size: typeof parsed.size === 'number' ? parsed.size : undefined,
   127→      }
   128→    }
   129→  } catch {
   130→    // Not JSON — fall through to plain-URL detection.
   131→  }
   132→  // Plain URL fallback (absolute http(s) or relative `/uploads/...`).
   133→  if (content.startsWith('http') || content.startsWith('/')) {
   134→    return { url: content }
   135→  }
   136→  return null
   137→}

   139→export function formatFileSize(bytes: number): string {
   140→  if (bytes < 1024) return `${bytes} B`
   141→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   142→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   143→}

   145→// Derive a filename from a URL when the parsed payload has no `name`.
   146→function deriveFilenameFromUrl(url: string, fallback: string): string {
   147→  try {
   148→    const u = new URL(url, typeof window !== 'undefined' ? window.location.origin : 'http://localhost')
   149→    const last = u.pathname.split('/').filter(Boolean).pop()
   150→    return last ? decodeURIComponent(last) : fallback
   151→  } catch {
   152→    return fallback
   153→  }
   154→}

   156→// ---- Protection helpers -------------------------------------------------
   157→// Resolves whether a message is effectively protected from screenshot /
   158→// forwarding. The effective flag is:
   159→//   - the explicit `message.protected` value if it is a boolean
   160→//   - otherwise the sender's `defaultProtectMessages` setting (we don't have
   161→//     the sender's settings here, so we just use the explicit value — the
   162→//     backend always resolves and persists it on send, so the frontend
   163→//     always sees the resolved boolean).
   164→//
   165→// Returns `{ isProtected, isOwner, alwaysAllow, blocked }` where `blocked`
   166→// is true only when the current user is the recipient of a protected
   167→// message and has NOT opted into "privacy always allow".
   168→function useProtectionState(message: ChatMessage) {
   169→  const me = useWaslStore((s) => s.user)
   170→  const isOwner = !!me && message.senderId === me.id
   171→  const alwaysAllow = !!me?.privacyAlwaysAllow
   172→  const isProtected = message.protected === true
   173→  const blocked = isProtected && !isOwner && !alwaysAllow
   174→  return { isProtected, isOwner, alwaysAllow, blocked }
   175→}

   177→// Record a screenshot/copy/save attempt to the audit log via the API and
   178→// show a toast. Idempotent — debounced per-message per-kind to avoid
   179→// spamming the server when the user holds down PrintScreen.
   180→const recentAttemptCache = new Map<string, number>()
   181→function recordAttempt(messageId: string, kind: string) {
   182→  const key = `${messageId}:${kind}`
   183→  const now = Date.now()
   184→  if (recentAttemptCache.has(key)) {
   185→    const last = recentAttemptCache.get(key)!
   186→    // Debounce: at most one record per 5 seconds per kind per message.
   187→    if (now - last < 5000) return
   188→  }
   189→  recentAttemptCache.set(key, now)
   190→  // Fire and forget
   191→  fetch(`/api/messages/${messageId}/screenshot-attempt`, {
   192→    method: 'POST',
   193→    headers: { 'Content-Type': 'application/json' },
   194→    body: JSON.stringify({ kind }),
   195→  }).catch(() => {})
   196→}

   989→// ---- File-attachment card (PDF / document) -------------------------------
   990→// Renders a compact "file card" inside the bubble: a coloured FileText icon
   991→// (red for PDF, teal/green for documents), the original filename + human-
   992→// readable size, and a Download button that opens the URL in a new tab.
   993→//
   994→// `blocked` is set when the message is protected AND the current user is the
   995→// recipient — in that case the Download button is disabled and shows the
   996→// protected warning toast instead of opening the link.
   997→function PdfDocumentCardContent({
   998→  message,
   999→  mine,
  1000→  blocked,
  1001→  isProtected,
  1002→  onOpenReadReceipts,
  1003→}: {
  1004→  message: ChatMessage
  1005→  mine: boolean
  1006→  blocked: boolean
  1007→  isProtected: boolean
  1008→  onOpenReadReceipts: () => void
  1009→}) {
  1010→  const parsed = parseFileContent(message.content)
  1011→  const isPdf = message.type === 'pdf'
  1012→  const fallbackName = isPdf ? 'PDF document' : 'Document'
  1013→  const name =
  1014→    parsed?.name ||
  1015→    (parsed?.url ? deriveFilenameFromUrl(parsed.url, fallbackName) : fallbackName)
  1016→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : isPdf ? 'PDF' : 'Document'
  1017→
  1018→  // PDF → red icon. Document → teal/green icon (uses wasl theme tokens so it
  1019→  // respects light/dark mode and the Cirkle gold theme). No indigo/blue.
  1020→  const iconColor = isPdf
  1021→    ? 'text-red-500'
  1022→    : 'text-[var(--wasl-teal)] dark:text-[var(--wasl-green)]'
  1023→  const iconBg = isPdf
  1024→    ? 'bg-red-500/10'
  1025→    : 'bg-[var(--wasl-teal)]/10 dark:bg-[var(--wasl-green)]/10'
  1026→
  1027→  return (
  1028→    <div className="rounded-lg overflow-hidden max-w-[280px]">
  1029→      <div className="flex items-center gap-3 p-2 min-w-[220px]">
  1030→        <div className={cn('w-10 h-10 rounded-lg flex items-center justify-center shrink-0', iconBg)}>
  1031→          <FileText className={cn('w-5 h-5', iconColor)} />
  1032→        </div>
  1033→        <div className="flex-1 min-w-0">
  1034→          <div className="text-sm font-medium truncate text-foreground" title={name}>
  1035→            {name}
  1036→          </div>
  1037→          <div className="text-[11px] text-foreground/60 uppercase tracking-wide">
  1038→            {sizeLabel}
  1039→          </div>
  1040→        </div>
  1041→        {blocked ? (
  1042→          <button
  1043→            type="button"
  1044→            onClick={(e) => {
  1045→              e.stopPropagation()
  1046→              toast.error(PROTECTED_WARNING, { duration: 4000 })
  1047→              recordAttempt(message.id, 'save')
  1048→            }}
  1049→            className="shrink-0 w-8 h-8 rounded-md flex items-center justify-center text-muted-foreground/40 cursor-not-allowed"
  1050→            title="Download disabled — message is protected"
  1051→            aria-label="Download disabled — message is protected"
  1052→          >
  1053→            <Download className="w-4 h-4" />
  1054→          </button>
  1055→        ) : (
  1056→          <a
  1057→            href={parsed?.url}
  1058→            target="_blank"
  1059→            rel="noopener noreferrer"
  1060→            download={name}
  1061→            onClick={(e) => e.stopPropagation()}
  1062→            className={cn(
  1063→              'shrink-0 px-2.5 py-1.5 rounded-md text-xs font-medium transition-colors',
  1064→              'bg-[var(--wasl-green)]/10 text-[var(--wasl-green)] hover:bg-[var(--wasl-green)]/20'
  1065→            )}
  1066→            title="Download"
  1067→            aria-label="Download"
  1068→          >
  1069→            <Download className="w-4 h-4" />
  1070→          </a>
  1071→        )}
  1072→      </div>
  1073→      <div className="text-[10px] text-right text-foreground/60 mt-0.5 flex items-center justify-end gap-1">
  1074→        {isProtected && <Lock className="w-3 h-3 inline opacity-60" />}
  1075→        {formatChatTimestamp(message.createdAt)}
  1076→        {mine && <StatusTicks status={message.status} className="ml-1" onClick={onOpenReadReceipts} />}

  1082→// ---- Audio URL card ------------------------------------------------------
  1083→// Renders an `<audio controls>` element for audio messages whose `content`
  1084→// is a URL (i.e. an uploaded .mp3/.wav/.m4a/.ogg). Voice notes that were
  1085→// recorded in-browser are stored as `data:audio/...` data URIs and are
  1086→// handled by the existing VoiceMessagePlayer above.
  1087→function AudioUrlCardContent({
  1088→  message,
  1089→  mine,
  1090→  blocked,
  1091→  isProtected,
  1092→  onOpenReadReceipts,
  1093→}: {
  1094→  message: ChatMessage
  1095→  mine: boolean
  1096→  blocked: boolean
  1097→  isProtected: boolean
  1098→  onOpenReadReceipts: () => void
  1099→}) {
  1100→  const parsed = parseFileContent(message.content)
  1101→  const url = parsed?.url
  1102→  const name =
  1103→    parsed?.name ||
  1104→    (url ? deriveFilenameFromUrl(url, 'Audio message') : 'Audio message')
  1105→  const sizeLabel = parsed?.size ? formatFileSize(parsed.size) : undefined
  1106→
  1107→  if (!url) {
  1108→    // Malformed payload — fall back to a tiny "audio unavailable" notice
  1109→    // rather than crashing the whole bubble.
  1110→    return (
  1111→      <div className="rounded-lg p-2 max-w-[280px] text-xs text-muted-foreground">
  1112→        Audio unavailable
  1113→      </div>
  1114→    )
  1115→  }
  1116→

  1118→    <div className="rounded-lg overflow-hidden max-w-[300px] p-1">

  1121→        <span className="text-xs font-medium truncate flex-1 text-foreground" title={name}>
  1122→          {name}
  1123→        </span>
  1124→        {sizeLabel && (
  1125→          <span className="text-[10px] text-foreground/60 shrink-0">{sizeLabel}</span>
  1126→        )}
  1127→        {blocked ? (
  1128→          <button
  1129→            type="button"
  1130→            onClick={(e) => {
  1131→              e.stopPropagation()
  1132→              toast.error(PROTECTED_WARNING, { duration: 4000 })
  1133→              recordAttempt(message.id, 'save')
  1134→            }}
  1135→            className="shrink-0 w-7 h-7 rounded-md flex items-center justify-center text-muted-foreground/40 cursor-not-allowed"
  1136→            title="Download disabled — message is protected"
  1137→            aria-label="Download disabled — message is protected"
  1138→          >
  1139→            <Download className="w-4 h-4" />

  1141→        ) : (
  1142→          <a
  1143→            href={url}
  1144→            target="_blank"
  1145→            rel="noopener noreferrer"
  1146→            download={name}
  1147→            onClick={(e) => e.stopPropagation()}
  1148→            className={cn(
  1149→              'shrink-0 px-2 py-1 rounded-md text-xs font-medium transition-colors',
  1150→              'bg-[var(--wasl-green)]/10 text-[var(--wasl-green)] hover:bg-[var(--wasl-green)]/20'

  1152→            title="Download"
  1153→            aria-label="Download"
  1154→          >
  1155→            <Download className="w-4 h-4" />
  1156→          </a>
  1157→        )}
  1158→      </div>
  1159→      {/* Custom audio player (Task 30-a) — replaces the bare `<audio controls>`
  1160→          element with the same reusable VoicePlayer used for voice notes.
  1161→          `variant="audio"` makes it show a Music icon and the surface tint
  1162→          matches the bubble (wasl-green for outgoing, wasl-teal for
  1163→          incoming). `blocked` makes the player read-only (pointer-events
  1164→          disabled) for protected recipients. */}
  1165→      <VoicePlayer
  1166→        src={url}
  1167→        mine={mine}
  1168→        variant="audio"
  1169→        blocked={blocked}
  1170→        label={name ? `Play audio: ${name}` : 'Play audio'}
  1171→      />
  1172→      <div className="text-[10px] text-right text-foreground/60 mt-0.5 flex items-center justify-end gap-1">
  1173→        {isProtected && <Lock className="w-3 h-3 inline opacity-60" />}
  1174→        {formatChatTimestamp(message.createdAt)}
  1175→        {mine && <StatusTicks status={message.status} className="ml-1" onClick={onOpenReadReceipts} />}
  1176→      </div>
  1177→    </div>
  1178→  )
  1179→}

  1210→function StatusTicks({
  1211→  status,
  1212→  className,
  1213→  onClick,
  1214→}: {
  1215→  status: string
  1216→  className?: string
  1217→  onClick?: () => void
  1218→}) {
  1219→  if (status === 'sent') {
  1220→    return <Check className={cn('w-3.5 h-3.5 inline', className)} />
  1221→  }
  1222→  if (status === 'delivered') {
  1223→    return <CheckCheck className={cn('w-3.5 h-3.5 inline', className)} />
  1224→  }
  1225→  if (status === 'read') {
  1226→    // When onClick is provided, the read-ticks become a clickable button that
  1227→    // opens the read-receipts dialog (sender-only feature).
  1228→    if (onClick) {
  1229→      return (
  1230→        <button
  1231→          type="button"
  1232→          onClick={(e) => {
  1233→            e.stopPropagation()
  1234→            onClick()

  1236→          className="inline-flex items-center hover:scale-110 transition-transform"
  1237→          title="Read — click to see who read this message"
  1238→          aria-label="Read — click to see details"

  1240→          <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />

  1243→    }
  1244→    return (
  1245→      <CheckCheck className={cn('w-3.5 h-3.5 inline text-sky-500', className)} />
  1246→    )
  1247→  }
  1248→  return <Clock className={cn('w-3.5 h-3.5 inline', className)} />

  1251→function groupReactions(reactions: Reaction[]) {
  1252→  const map = new Map<string, { emoji: string; count: number; users: string[] }>()
  1253→  for (const r of reactions) {
  1254→    const ex = map.get(r.emoji)
  1255→    if (ex) {
  1256→      ex.count++
  1257→      ex.users.push(r.userId)
  1258→    } else {
  1259→      map.set(r.emoji, { emoji: r.emoji, count: 1, users: [r.userId] })
  1260→    }
  1261→  }
  1262→  return Array.from(map.values()).sort((a, b) => b.count - a.count)
  1263→}
