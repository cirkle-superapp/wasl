   451→// Official Service Providers — banks, government ministries, utilities.
   452→// These are manually authenticated and approved by Wasl admin staff.
   453→// Once approved, they can broadcast official announcements to all users
   454→// within a specific country.
   455→//
   456→// Registration flow:
   457→//   1. User applies → uploads official registration documents + passes
   458→//      liveness test (selfie video) + ID document
   459→//   2. Wasl admin manually reviews → approves or rejects
   460→//   3. Once approved, the provider gets a verified badge + broadcast rights
   461→//
   462→// Broadcast flow:
   463→//   1. Provider creates an announcement (text/image)
   464→//   2. System delivers to ALL users whose phone country code matches
   465→//   3. Users see official announcements in a dedicated "Official" tab
   466→// ==========================================================================
   467→
   468→model ServiceProvider {
   469→  id                String   @id @default(cuid())
   470→  name              String   // e.g. "Ministry of Health", "National Bank of Egypt"
   471→  ownerId           String   // the user who registered the provider
   472→  type              String   @default("government") // government | bank | utility | telecom | healthcare
   473→  description       String   @default("")
   474→  avatarPath        String?
   475→  avatarColor       String?
   476→  // Official registration details
   477→  officialName      String?  // legal entity name
   478→  registrationNumber String? // government registration number
   479→  taxNumber         String?
   480→  // Country code for broadcast targeting (e.g. "+20" for Egypt)
   481→  // All users whose phone starts with this code receive broadcasts
   482→  countryCode       String   @default("+20")
   483→  // Verification documents
   484→  registrationDocPath String?  // official registration document
   485→  idDocPath           String?  // ID of the authorized representative
   486→  livenessVideoPath  String?  // selfie video for liveness check
   487→  livenessVerified   Boolean  @default(false)
   488→  // Approval state — manually reviewed by Wasl admin
   489→  status            String   @default("pending") // pending | approved | rejected
   490→  rejectionReason   String?
   491→  verified          Boolean  @default(false)
   492→  verifiedAt        DateTime?
   493→  verifiedBy        String?  // admin user ID who approved
   494→  // Broadcast settings
   495→  canBroadcast      Boolean  @default(false)  // can send announcements
   496→  // Contact info
   497→  contactPhone      String?
   498→  contactEmail      String?
   499→  website           String?
   500→  createdAt         DateTime @default(now())
   501→  updatedAt         DateTime @updatedAt
   502→
   503→  owner     User                  @relation("ServiceProviderOwner", fields: [ownerId], references: [id], onDelete: Cascade)
   504→  broadcasts ServiceProviderMessage[]
   505→
   506→  @@index([ownerId])
   507→  @@index([countryCode])
   508→  @@index([status])
   509→  @@index([type])

   561→/// Scheduled message — send at a future time (Cirkle-inspired scheduled-messages)
   562→/// Supports recurring schedules: `repeat` can be 'none' (one-time), 'daily',
   563→/// 'weekly', or 'monthly'. When a recurring message is sent, `scheduledFor`
   564→/// is advanced to the next occurrence and `sent`/`sentAt` are reset so the
   565→/// background scheduler picks it up again at the next time.
   566→model ScheduledMessage {
   567→  id             String   @id @default(cuid())
   568→  conversationId String
   569→  senderId       String
   570→  content        String
   571→  type           String   @default("text")
   572→  scheduledFor   DateTime
   573→  sent           Boolean  @default(false)
   574→  sentAt         DateTime?
   575→  // Recurring schedule: 'none' = one-time, 'daily', 'weekly', 'monthly'.
   576→  // When non-'none', the scheduler advances `scheduledFor` to the next
