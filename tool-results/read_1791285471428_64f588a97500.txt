   145→  async function switchPortal(id: string) {
   146→    setPortals((cur) => cur.map((p) => ({ ...p, active: p.id === id })))
   147→    try {
   148→      const res = await fetch(`/api/phone-numbers/${id}`, {
   149→        method: 'PATCH',
   150→        headers: { 'Content-Type': 'application/json' },
   151→        body: JSON.stringify({ active: true }),
   152→      })
   153→      if (!res.ok) {
   154→        const d = await res.json()
   155→        toast.error(d?.error || 'Failed to switch portal')
   156→        // Reload to restore the server's truth
   157→        const fresh = await fetch('/api/phone-numbers', { cache: 'no-store' })
   158→        if (fresh.ok) {
   159→          const data = await fresh.json()
   160→          setPortals(data.phoneNumbers || [])
   161→        }
   162→        return
   163→      }
   164→      const data = await res.json()
   165→      // Sync the user.phone field so the sidebar shows the new active number
   166→      if (user) {
   167→        setUser({ ...user, phone: data.number, phoneNumbers: portals.map((p) => ({ ...p, active: p.id === id })) })
   168→      }
   169→      toast.success(`Switched to ${data.portalName || 'default'} portal`)
   170→    } catch {
   171→      toast.error('Network error')
   172→    }

   232→  function togglePin(id: string) {
   233→    setPinnedIds((prev) => {
   234→      const next = new Set(prev)
   235→      if (next.has(id)) {
   236→        next.delete(id)
   237→        toast.success('Conversation unpinned')
   238→      } else {
   239→        next.add(id)
   240→        toast.success('Conversation pinned to top')
   241→      }
   242→      try { localStorage.setItem('wasl-pinned-conversations', JSON.stringify([...next])) } catch {}
   243→      return next
   244→    })
   245→  }

   247→  // Sort: pinned conversations first, then by last message time

   250→    const bPinned = pinnedIds.has(b.id) ? 1 : 0

   721→function ConversationRow({
   722→  conversation,
   723→  active,
   724→  onlineUserIds,
   725→  pinned,
   726→  onPin,
   727→  onClick,
   728→  onDelete,
   729→  onArchive,
   730→  onMarkUnread,
   731→  index = 0,
   732→}: {
   733→  conversation: Conversation
   734→  active: boolean
   735→  onlineUserIds: Set<string>
   736→  pinned?: boolean
   737→  onPin?: () => void
   738→  onClick: () => void
   739→  onDelete: () => void
   740→  onArchive: () => void
   741→  onMarkUnread?: () => void
   742→  index?: number
   743→}) {
   744→  const last = conversation.lastMessage
   745→  const otherUser = !conversation.isGroup
   746→    ? conversation.participants.find((p) => p.userId !== useWaslStore.getState().user?.id)
   747→    : null
   748→  const isOnline =
   749→    !conversation.isGroup &&
   750→    !!otherUser &&
   751→    (onlineUserIds.has(otherUser.userId) || otherUser.online)
   752→
   753→  // Reactive subscription to typing state for THIS conversation.
   754→  // Select only the raw typing object (not a transformed array) so Zustand
   755→  // can cache the reference and avoid infinite re-renders. The transformation
   756→  // to display names is done in the component body below.
   757→  const typingByConv = useWaslStore((s) => s.typingByConversation[conversation.id])
   758→  const myId = useWaslStore((s) => s.user?.id)
   759→  // Subscribe to the draft text for THIS conversation so the "Draft" badge
   760→  // + "Draft: <preview>" last-message text react in real-time as the user
   761→  // types or sends.
   762→  const draftText = useWaslStore((s) => s.drafts[conversation.id] || '')
   763→  const hasDraft = draftText.trim().length > 0
   764→  const typingNames = typingByConv
   765→    ? Object.entries(typingByConv)
   766→        .filter(([uid]) => uid !== myId)
   767→        .map(([uid]) => {
   768→          const p = conversation.participants.find((pp) => pp.userId === uid)
   769→          return p?.name?.split(' ')[0] || 'Someone'
   770→        })
   771→    : []

   773→
   774→  const isMine = last?.senderId === useWaslStore.getState().user?.id
   775→

   777→  if (last?.type === 'image') preview = '📷 Photo'
   778→  if (last?.type === 'pdf') preview = '📄 PDF'
   779→  if (last?.type === 'document') preview = '📄 Document'
   780→  if (last?.type === 'audio') preview = '🎵 Audio'

   785→  // shows "Draft: <text>" in the conversation row when there's an unsent
   786→  // message saved for that conversation.

   788→
   789→  const previewSender =
   790→    conversation.isGroup && last && !isMine && last.type !== 'system'
   791→      ? `${conversation.participants.find((p) => p.userId === last.senderId)?.name?.split(' ')[0] || 'Someone'}: `
   792→      : isMine && last && last.type !== 'system'
   793→      ? 'You: '
   794→      : ''
   795→
   796→  // Typing preview text — "typing…" for 1-on-1, "Name is typing…" for groups
   797→  const typingText =
   798→    conversation.isGroup && typingNames.length === 1
   799→      ? `${typingNames[0]} is typing…`
   800→      : conversation.isGroup && typingNames.length > 1
   801→        ? `${typingNames.length} people typing…`
   802→        : 'typing…'
   803→
   804→  return (
   805→    <div
   806→      onClick={onClick}
   807→      className={cn(
   808→        'group relative flex items-center gap-3 cursor-pointer border-b border-border/60',
   809→        'wasl-conv-card wasl-conv-row',
   810→        index < 6 && `wasl-anim-slide-up wasl-stagger-${index + 1}`,
   811→        active
   812→          ? 'wasl-conv-card-active'
   813→          : ''
   814→      )}
   815→    >
   816→      {/* Selected conversation: gradient accent bar — rendered by the
   817→          .wasl-active-accent class via ::before (green→teal gradient with
   818→          a soft glow). The previous solid-color span is removed in favor
   819→          of this premium treatment. */}
   820→      {conversation.isGroup ? (
   821→        <WaslGroupAvatar
   822→          name={conversation.name}
   823→          src={conversation.avatar}
   824→          participants={conversation.participants.map((p) => ({
   825→            name: p.name,
   826→            avatar: p.avatar,
   827→            avatarColor: p.avatarColor,
   828→          }))}
   829→          size={48}
   830→        />
   831→      ) : (
   832→        <WaslAvatar
   833→          name={conversation.name}
   834→          src={conversation.avatar}
   835→          color={conversation.avatarColor}
   836→          size={48}
   837→          online={isOnline}
   838→          showStatus
   839→        />
   840→      )}
   841→      <div className="flex-1 min-w-0">
   842→        <div className="flex items-center justify-between gap-2">
   843→          <div
   844→            className={cn(
   845→              'wasl-conv-name truncate text-foreground flex items-center gap-1.5',
   846→              conversation.unreadCount > 0 && 'wasl-conv-name-unread'
   847→            )}
   848→          >
   849→            <span className="truncate">{conversation.name}</span>
   850→            {conversation.unreadCount > 0 && (
   851→              <span className="wasl-unread-dot" />
   852→            )}
   853→            {pinned && (
   854→              <Pin className="w-3 h-3 text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] shrink-0" fill="currentColor" />
   855→            )}
   856→            {/* Subtle "Draft" badge — shown only when there's unsent text for
   857→                this conversation. Uses italic muted-foreground to match
   858→                WhatsApp's quiet draft indicator. */}
   859→            {hasDraft && (
   860→              <span
   861→                title="You have an unsent draft in this conversation"
   862→                className="inline-flex items-center gap-0.5 text-[10px] italic font-medium text-[var(--wasl-teal)] dark:text-[var(--wasl-green)] bg-[var(--wasl-green)]/10 rounded px-1 py-0.5 shrink-0"

   872→                className={cn(

   874→                  conversation.unreadCount > 0

   885→          <div
   886→            className={cn(
   887→              'wasl-conv-preview flex items-center gap-1',
   888→              isTyping
